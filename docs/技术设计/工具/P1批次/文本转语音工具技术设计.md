# 文本转语音工具技术设计文档

## 一、概述

本文档详细描述MaxTool 0.2版本中文本转语音工具（Text to Speech）的技术设计，包括功能设计、技术实现、组件结构和代码复用等方面。设计严格遵循项目规范，注重代码复用、组件化和性能优化。

## 二、功能设计

### 1. 核心功能

#### 文本转语音
- 支持将输入的文本转换为语音
- 支持多种语言和方言选择
- 支持多种声音（男声、女声、中性）选择
- 支持调整语速和音调

#### 语音控制
- 支持播放、暂停、停止语音
- 支持进度控制
- 支持音量调整

#### 辅助功能
- 支持复制生成的语音链接
- 支持下载生成的语音文件
- 支持清空输入

### 2. 用户界面

#### 文本输入区域
- 多行文本输入框：用于输入要转换的文本
- 字符计数：显示当前输入的字符数
- 清空按钮：快速清空输入内容

#### 语音设置
- 语言选择器：选择语音的语言和方言
- 声音选择器：选择语音的声音类型
- 语速滑块：调整语音的播放速度
- 音调滑块：调整语音的音调
- 音量滑块：调整语音的播放音量

#### 控制区域
- 播放按钮：播放生成的语音
- 暂停按钮：暂停语音播放
- 停止按钮：停止语音播放
- 进度条：显示语音播放进度

#### 输出区域
- 语音预览：显示语音播放状态
- 下载按钮：下载生成的语音文件
- 复制按钮：复制语音链接

#### 使用说明
- 提供简洁的使用步骤说明

## 三、技术设计

### 1. 技术选型

#### 核心库
- Web Speech API：用于浏览器内的文本转语音功能

#### 浏览器API
- SpeechSynthesis API：用于语音合成
- Clipboard API：用于复制语音链接
- MediaRecorder API：用于录制和下载语音

### 2. 组件结构

#### 主组件

##### TextToSpeech
- **位置**：`components/tools/TextToSpeech.tsx`
- **功能**：文本转语音工具的主组件
- **结构**：
  ```tsx
  'use client'

  import { useState, useEffect, useRef } from 'react'
  import Button from '@/components/ui/Button'
  import Textarea from '@/components/ui/Textarea'
  import Select from '@/components/ui/Select'
  import Slider from '@/components/ui/Slider'
  import { trackUsage } from '@/lib/tracking'
  import UsageInstructions from '@/components/shared/UsageInstructions'
  import { copyToClipboard } from '@/lib/clipboard-utils'
  import { getVoices } from '@/lib/speech-utils'

  export default function TextToSpeech() {
    // State declarations
    const [text, setText] = useState('')
    const [language, setLanguage] = useState('zh-CN')
    const [voice, setVoice] = useState<string>('')
    const [rate, setRate] = useState(1.0)
    const [pitch, setPitch] = useState(1.0)
    const [volume, setVolume] = useState(1.0)
    const [isPlaying, setIsPlaying] = useState(false)
    const [isPaused, setIsPaused] = useState(false)
    const [currentTime, setCurrentTime] = useState(0)
    const [duration, setDuration] = useState(0)
    const [voices, setVoices] = useState<SpeechSynthesisVoice[]>([])
    const [hasTracked, setHasTracked] = useState(false)

    // Refs
    const utteranceRef = useRef<SpeechSynthesisUtterance | null>(null)
    const speechSynthesisRef = useRef<SpeechSynthesis | null>(null)
    const progressIntervalRef = useRef<NodeJS.Timeout | null>(null)

    // Initialize speech synthesis
    useEffect(() => {
      if ('speechSynthesis' in window) {
        speechSynthesisRef.current = window.speechSynthesis
        const availableVoices = getVoices()
        setVoices(availableVoices)
        
        // Set default voice
        if (availableVoices.length > 0) {
          const defaultVoice = availableVoices.find(v => v.lang === 'zh-CN' && v.name.includes('Female'))
          if (defaultVoice) {
            setVoice(defaultVoice.voiceURI)
          } else {
            setVoice(availableVoices[0].voiceURI)
          }
        }
      }

      // Clean up
      return () => {
        if (progressIntervalRef.current) {
          clearInterval(progressIntervalRef.current)
        }
      }
    }, [])

    // Update voices when language changes
    useEffect(() => {
      const filteredVoices = voices.filter(v => v.lang.startsWith(language))
      if (filteredVoices.length > 0) {
        setVoice(filteredVoices[0].voiceURI)
      }
    }, [language, voices])

    // Event handlers
    const handleTextChange = (e: React.ChangeEvent<HTMLTextAreaElement>) => {
      setText(e.target.value)
    }

    const handlePlay = () => {
      if (!text.trim()) return
      
      // Track usage
      if (!hasTracked) {
        trackUsage('text-to-speech')
        setHasTracked(true)
      }

      if (isPaused && utteranceRef.current) {
        // Resume paused speech
        speechSynthesisRef.current?.resume()
        setIsPlaying(true)
        setIsPaused(false)
      } else {
        // Cancel any ongoing speech
        speechSynthesisRef.current?.cancel()

        // Create new utterance
        const utterance = new SpeechSynthesisUtterance(text)
        utteranceRef.current = utterance

        // Configure utterance
        const selectedVoice = voices.find(v => v.voiceURI === voice)
        if (selectedVoice) {
          utterance.voice = selectedVoice
        }
        utterance.rate = rate
        utterance.pitch = pitch
        utterance.volume = volume

        // Set up event listeners
        utterance.onstart = () => {
          setIsPlaying(true)
          setIsPaused(false)
          setCurrentTime(0)
          // Estimate duration based on text length and rate
          setDuration(text.length / (rate * 10))
          
          // Start progress tracking
          if (progressIntervalRef.current) {
            clearInterval(progressIntervalRef.current)
          }
          progressIntervalRef.current = setInterval(() => {
            setCurrentTime(prev => {
              if (prev >= duration) {
                clearInterval(progressIntervalRef.current!)
                return duration
              }
              return prev + 0.1
            })
          }, 100)
        }

        utterance.onpause = () => {
          setIsPlaying(false)
          setIsPaused(true)
          if (progressIntervalRef.current) {
            clearInterval(progressIntervalRef.current)
          }
        }

        utterance.onresume = () => {
          setIsPlaying(true)
          setIsPaused(false)
          if (progressIntervalRef.current) {
            clearInterval(progressIntervalRef.current)
          }
          progressIntervalRef.current = setInterval(() => {
            setCurrentTime(prev => {
              if (prev >= duration) {
                clearInterval(progressIntervalRef.current!)
                return duration
              }
              return prev + 0.1
            })
          }, 100)
        }

        utterance.onend = () => {
          setIsPlaying(false)
          setIsPaused(false)
          setCurrentTime(duration)
          if (progressIntervalRef.current) {
            clearInterval(progressIntervalRef.current)
          }
        }

        utterance.onerror = (event) => {
          console.error('Speech synthesis error:', event.error)
          setIsPlaying(false)
          setIsPaused(false)
          if (progressIntervalRef.current) {
            clearInterval(progressIntervalRef.current)
          }
        }

        // Speak the utterance
        speechSynthesisRef.current?.speak(utterance)
      }
    }

    const handlePause = () => {
      speechSynthesisRef.current?.pause()
    }

    const handleStop = () => {
      speechSynthesisRef.current?.cancel()
      setIsPlaying(false)
      setIsPaused(false)
      setCurrentTime(0)
      if (progressIntervalRef.current) {
        clearInterval(progressIntervalRef.current)
      }
    }

    const handleProgressChange = (e: React.ChangeEvent<HTMLInputElement>) => {
      const newTime = parseFloat(e.target.value)
      setCurrentTime(newTime)
      // Note: Web Speech API doesn't support seeking, so we can't actually change the playback position
    }

    const handleDownload = () => {
      // This would require additional implementation using MediaRecorder API
      // For simplicity, we'll just log a message for now
      console.log('Download functionality would be implemented here')
    }

    const handleCopy = () => {
      // This would require generating a shareable link to the speech
      // For simplicity, we'll just copy the text for now
      copyToClipboard(text)
    }

    const handleClear = () => {
      setText('')
      handleStop()
    }

    // Render
    return (
      <div className="space-y-6">
        {/* Text Input Section */}
        <div className="space-y-4">
          <h2 className="text-xl font-semibold">文本输入</h2>
          <div className="relative">
            <Textarea
              value={text}
              onChange={handleTextChange}
              placeholder="请输入要转换为语音的文本..."
              className="w-full h-40"
            />
            <div className="absolute bottom-2 right-2 text-sm text-gray-500">
              {text.length} 字符
            </div>
          </div>
          <Button onClick={handleClear} variant="outline" size="sm">
            清空
          </Button>
        </div>

        {/* Voice Settings Section */}
        <div className="space-y-4">
          <h2 className="text-xl font-semibold">语音设置</h2>
          
          <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
            <div className="space-y-2">
              <label className="block">语言</label>
              <Select
                value={language}
                onChange={(e) => setLanguage(e.target.value)}
                className="w-full"
              >
                <option value="zh-CN">中文（中国大陆）</option>
                <option value="zh-TW">中文（台湾）</option>
                <option value="en-US">英语（美国）</option>
                <option value="en-GB">英语（英国）</option>
                <option value="ja-JP">日语</option>
                <option value="ko-KR">韩语</option>
                <option value="fr-FR">法语</option>
                <option value="de-DE">德语</option>
                <option value="es-ES">西班牙语</option>
                <option value="it-IT">意大利语</option>
              </Select>
            </div>
            
            <div className="space-y-2">
              <label className="block">声音</label>
              <Select
                value={voice}
                onChange={(e) => setVoice(e.target.value)}
                className="w-full"
              >
                {voices
                  .filter(v => v.lang.startsWith(language))
                  .map(v => (
                    <option key={v.voiceURI} value={v.voiceURI}>
                      {v.name} ({v.lang})
                    </option>
                  ))
                }
              </Select>
            </div>
          </div>

          <div className="space-y-2">
            <div className="flex justify-between">
              <label className="block">语速</label>
              <span className="text-sm text-gray-500">{rate.toFixed(1)}x</span>
            </div>
            <Slider
              value={[rate]}
              onChange={(value) => setRate(value[0])}
              min={0.1}
              max={3}
              step={0.1}
            />
          </div>

          <div className="space-y-2">
            <div className="flex justify-between">
              <label className="block">音调</label>
              <span className="text-sm text-gray-500">{pitch.toFixed(1)}</span>
            </div>
            <Slider
              value={[pitch]}
              onChange={(value) => setPitch(value[0])}
              min={0}
              max={2}
              step={0.1}
            />
          </div>

          <div className="space-y-2">
            <div className="flex justify-between">
              <label className="block">音量</label>
              <span className="text-sm text-gray-500">{Math.round(volume * 100)}%</span>
            </div>
            <Slider
              value={[volume]}
              onChange={(value) => setVolume(value[0])}
              min={0}
              max={1}
              step={0.01}
            />
          </div>
        </div>

        {/* Control Section */}
        <div className="space-y-4">
          <h2 className="text-xl font-semibold">控制</h2>
          
          <div className="flex gap-4">
            <Button
              onClick={handlePlay}
              variant="primary"
              size="lg"
              disabled={!text.trim()}
            >
              {isPaused ? '继续' : '播放'}
            </Button>
            <Button
              onClick={handlePause}
              variant="secondary"
              size="lg"
              disabled={!isPlaying}
            >
              暂停
            </Button>
            <Button
              onClick={handleStop}
              variant="outline"
              size="lg"
              disabled={!isPlaying && !isPaused}
            >
              停止
            </Button>
          </div>

          {isPlaying && duration > 0 && (
            <div className="space-y-2">
              <div className="flex justify-between">
                <span className="text-sm text-gray-500">进度</span>
                <span className="text-sm text-gray-500">
                  {currentTime.toFixed(1)} / {duration.toFixed(1)} 秒
                </span>
              </div>
              <input
                type="range"
                min="0"
                max={duration}
                value={currentTime}
                onChange={handleProgressChange}
                className="w-full"
              />
            </div>
          )}
        </div>

        {/* Output Section */}
        <div className="space-y-4">
          <h2 className="text-xl font-semibold">输出</h2>
          
          <div className="flex gap-4">
            <Button
              onClick={handleDownload}
              variant="secondary"
              size="lg"
              disabled={!text.trim()}
            >
              下载语音
            </Button>
            <Button
              onClick={handleCopy}
              variant="outline"
              size="lg"
              disabled={!text.trim()}
            >
              复制链接
            </Button>
          </div>
        </div>

        {/* Usage Instructions */}
        <UsageInstructions
          steps={[
            '在文本输入框中输入要转换为语音的文本',
            '选择语音的语言、声音类型、语速和音调',
            '点击"播放"按钮听取生成的语音',
            '使用暂停和停止按钮控制语音播放',
            '调整音量滑块控制语音播放音量',
            '点击"下载语音"按钮保存生成的语音文件',
            '点击"复制链接"按钮复制语音链接'
          ]}
        />
      </div>
    )
  }
  ```
