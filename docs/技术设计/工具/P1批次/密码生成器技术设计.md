# 密码生成器技术设计文档

## 一、概述

本文档详细描述MaxTool 0.2版本中密码生成器（Password Generator）工具的技术设计，包括功能设计、技术实现、组件结构和代码复用等方面。设计严格遵循项目规范，注重代码复用、组件化和性能优化。

## 二、功能设计

### 1. 核心功能

#### 基本密码生成
- 生成随机、安全的密码
- 支持自定义密码长度
- 支持多种字符类型选择

#### 高级功能
- 字符类型选择：
  - 大写字母（A-Z）
  - 小写字母（a-z）
  - 数字（0-9）
  - 特殊符号（!@#$%^&*等）
- 排除相似字符：如1和l，0和O
- 排除特定字符：允许用户指定不包含的字符
- 生成多个密码：一次生成多个密码供选择

### 2. 用户界面

#### 密码设置区域
- 长度滑块：用于调整密码长度（8-128个字符）
- 字符类型选项：勾选需要包含的字符类型
- 高级选项：
  - 排除相似字符复选框
  - 排除字符输入框
  - 生成密码数量选择

#### 密码显示区域
- 生成的密码显示：以密码形式显示（可切换可见性）
- 密码强度指示器：显示密码的强度评分（弱、中、强、极强）
- 复制按钮：复制生成的密码

#### 使用说明
- 提供简洁的使用步骤说明

#### 浏览器API

- Crypto API：用于生成安全的随机数

### 2. 组件结构

#### 主组件

##### PasswordGenerator
- **位置**：`components/tools/PasswordGenerator.tsx`
- **功能**：密码生成器工具的主组件
- **结构**：
  ```tsx
  'use client'

  import { useState, useEffect } from 'react'
  import Button from '@/components/ui/Button'
  import Input from '@/components/ui/Input'
  import { trackUsage } from '@/lib/tracking'
  import UsageInstructions from '@/components/shared/UsageInstructions'
  import { copyToClipboard } from '@/lib/clipboard-utils'

  // 定义字符类型选项
  interface CharacterTypeOptions {
    uppercase: boolean;
    lowercase: boolean;
    numbers: boolean;
    symbols: boolean;
  }

  // 定义密码强度等级
  type PasswordStrength = 'weak' | 'medium' | 'strong' | 'very-strong'

  // 相似字符列表
  const SIMILAR_CHARS = 'il1Lo0O'

  // 特殊符号列表
  const SYMBOLS = '!"#$%&\'()*+,-./:;<=>?@[\\]^_`{|}~'

  export default function PasswordGenerator() {
    // State declarations
    const [passwordLength, setPasswordLength] = useState(16)
    const [characterTypes, setCharacterTypes] = useState<CharacterTypeOptions>({
      uppercase: true,
      lowercase: true,
      numbers: true,
      symbols: true
    })
    const [excludeSimilar, setExcludeSimilar] = useState(false)
    const [excludeCharacters, setExcludeCharacters] = useState('')
    const [passwordCount, setPasswordCount] = useState(1)
    const [passwords, setPasswords] = useState<string[]>([''])
    const [showPassword, setShowPassword] = useState(false)
    const [passwordStrength, setPasswordStrength] = useState<PasswordStrength>('weak')
    const [error, setError] = useState<string | null>(null)
    const [hasTracked, setHasTracked] = useState(false)

    // 初始化
    useEffect(() => {
      trackUsage('password-generator')
      setHasTracked(true)
    }, [])

    // 生成密码
    useEffect(() => {
      generatePasswords()
    }, [passwordLength, characterTypes, excludeSimilar, excludeCharacters, passwordCount])

    // 生成密码核心逻辑
    const generatePasswords = () => {
      // 验证至少选择一种字符类型
      if (!characterTypes.uppercase && !characterTypes.lowercase && !characterTypes.numbers && !characterTypes.symbols) {
        setError('请至少选择一种字符类型')
        setPasswords([''])
        setPasswordStrength('weak')
        return
      }

      try {
        const generatedPasswords: string[] = []

        for (let i = 0; i < passwordCount; i++) {
          const password = generateSinglePassword()
          generatedPasswords.push(password)
        }

        setPasswords(generatedPasswords)
        setError(null)

        // 更新密码强度
        if (generatedPasswords.length > 0) {
          setPasswordStrength(calculatePasswordStrength(generatedPasswords[0]))
        }
      } catch (err) {
        setError(err instanceof Error ? err.message : '密码生成失败')
        setPasswords([''])
        setPasswordStrength('weak')
      }
    }

    // 生成单个密码
    const generateSinglePassword = (): string => {
      let characterSet = ''

      // 构建字符集
      if (characterTypes.uppercase) characterSet += 'ABCDEFGHIJKLMNOPQRSTUVWXYZ'
      if (characterTypes.lowercase) characterSet += 'abcdefghijklmnopqrstuvwxyz'
      if (characterTypes.numbers) characterSet += '0123456789'
      if (characterTypes.symbols) characterSet += SYMBOLS

      // 排除相似字符
      if (excludeSimilar) {
        characterSet = characterSet.split('').filter(char => !SIMILAR_CHARS.includes(char)).join('')
      }

      // 排除特定字符
      if (excludeCharacters) {
        const excludeSet = new Set(excludeCharacters)
        characterSet = characterSet.split('').filter(char => !excludeSet.has(char)).join('')
      }

      // 确保字符集不为空
      if (!characterSet) {
        throw new Error('字符集为空，请调整设置')
      }

      // 生成密码
      let password = ''
      for (let i = 0; i < passwordLength; i++) {
        const randomIndex = getSecureRandomIndex(characterSet.length)
        password += characterSet[randomIndex]
      }

      // 验证密码至少包含选择的每种字符类型（如果选择了多种）
      if (!isPasswordValid(password)) {
        // 递归生成新密码
        return generateSinglePassword()
      }

      return password
    }

    // 获取安全的随机索引
    const getSecureRandomIndex = (max: number): number => {
      const randomBytes = new Uint8Array(1)
      window.crypto.getRandomValues(randomBytes)
      return Math.floor((randomBytes[0] / 255) * max)
    }

    // 验证密码是否有效
    const isPasswordValid = (password: string): boolean => {
      // 如果只选择了一种字符类型，直接返回true
      const selectedTypesCount = Object.values(characterTypes).filter(Boolean).length
      if (selectedTypesCount <= 1) {
        return true
      }

      let isValid = true

      if (characterTypes.uppercase && !/[A-Z]/.test(password)) {
        isValid = false
      }
      if (characterTypes.lowercase && !/[a-z]/.test(password)) {
        isValid = false
      }
      if (characterTypes.numbers && !/[0-9]/.test(password)) {
        isValid = false
      }
      if (characterTypes.symbols && !new RegExp(`[${SYMBOLS.replace(/[.*+?^${}()|[\]\\]/g, '\\$&')}]`).test(password)) {
        isValid = false
      }

      return isValid
    }

    // 计算密码强度
    const calculatePasswordStrength = (password: string): PasswordStrength => {
      // 基于长度和字符类型计算强度
      const score = passwordLength * 0.5
      
      if (characterTypes.uppercase) score += 1
      if (characterTypes.lowercase) score += 1
      if (characterTypes.numbers) score += 1
      if (characterTypes.symbols) score += 2

      if (score < 12) return 'weak'
      if (score < 20) return 'medium'
      if (score < 28) return 'strong'
      return 'very-strong'
    }

    // 获取密码强度显示文本
    const getPasswordStrengthText = (strength: PasswordStrength): string => {
      switch (strength) {
        case 'weak': return '弱'
        case 'medium': return '中'
        case 'strong': return '强'
        case 'very-strong': return '极强'
      }
    }

    // 获取密码强度颜色
    const getPasswordStrengthColor = (strength: PasswordStrength): string => {
      switch (strength) {
        case 'weak': return 'text-red-500 bg-red-100'
        case 'medium': return 'text-yellow-500 bg-yellow-100'
        case 'strong': return 'text-green-500 bg-green-100'
        case 'very-strong': return 'text-blue-500 bg-blue-100'
      }
    }

    // 复制密码
    const handleCopyPassword = (password: string) => {
      if (password) {
        copyToClipboard(password)
        // 显示复制成功提示
      }
    }

    // 重新生成密码
    const handleRegenerate = () => {
      generatePasswords()
    }

    // Event handlers
    const handleCharacterTypeChange = (type: keyof CharacterTypeOptions) => {
      setCharacterTypes(prev => ({
        ...prev,
        [type]: !prev[type]
      }))
    }

    // Render
    return (
      <div className="space-y-6">
        {/* Header */}
        <div className="space-y-4">
          <h2 className="text-xl font-semibold">密码生成器</h2>
        </div>

        {/* Password Settings */}
        <div className="space-y-6">
          {/* Length Setting */}
          <div className="space-y-4">
            <h3 className="text-lg font-semibold">密码设置</h3>
            
            <div className="space-y-2">
              <div className="flex justify-between items-center">
                <label className="block">密码长度</label>
                <span className="text-lg font-medium">{passwordLength}</span>
              </div>
              <input
                type="range"
                min="8"
                max="128"
                value={passwordLength}
                onChange={(e) => setPasswordLength(parseInt(e.target.value))}
                className="w-full"
              />
              <div className="flex justify-between text-xs text-text-secondary">
                <span>8</span>
                <span>64</span>
                <span>128</span>
              </div>
            </div>
          </div>

          {/* Character Types */}
          <div className="space-y-4">
            <h3 className="text-lg font-semibold">字符类型</h3>
            
            <div className="grid grid-cols-2 gap-4">
              <div className="flex items-center space-x-2">
                <input
                  type="checkbox"
                  id="uppercase"
                  checked={characterTypes.uppercase}
                  onChange={() => handleCharacterTypeChange('uppercase')}
                />
                <label htmlFor="uppercase">大写字母 (A-Z)</label>
              </div>
              
              <div className="flex items-center space-x-2">
                <input
                  type="checkbox"
                  id="lowercase"
                  checked={characterTypes.lowercase}
                  onChange={() => handleCharacterTypeChange('lowercase')}
                />
                <label htmlFor="lowercase">小写字母 (a-z)</label>
              </div>
              
              <div className="flex items-center space-x-2">
                <input
                  type="checkbox"
                  id="numbers"
                  checked={characterTypes.numbers}
                  onChange={() => handleCharacterTypeChange('numbers')}
                />
                <label htmlFor="numbers">数字 (0-9)</label>
              </div>
              
              <div className="flex items-center space-x-2">
                <input
                  type="checkbox"
                  id="symbols"
                  checked={characterTypes.symbols}
                  onChange={() => handleCharacterTypeChange('symbols')}
                />
                <label htmlFor="symbols">特殊符号 (!@#$%^&*)</label>
              </div>
            </div>
          </div>

          {/* Advanced Options */}
          <div className="space-y-4">
            <h3 className="text-lg font-semibold">高级选项</h3>
            
            <div className="space-y-4">
              <div className="flex items-center space-x-2">
                <input
                  type="checkbox"
                  id="excludeSimilar"
                  checked={excludeSimilar}
                  onChange={() => setExcludeSimilar(!excludeSimilar)}
                />
                <label htmlFor="excludeSimilar">排除相似字符 (如 1l0O)</label>
              </div>
              
              <div className="space-y-2">
                <label htmlFor="excludeCharacters">排除特定字符</label>
                <Input
                  type="text"
                  id="excludeCharacters"
                  value={excludeCharacters}
                  onChange={(e) => setExcludeCharacters(e.target.value)}
                  placeholder="例如: abc123"
                  className="w-full"
                />
              </div>
              
              <div className="space-y-2">
                <label htmlFor="passwordCount">生成密码数量</label>
                <select
                  id="passwordCount"
                  value={passwordCount}
                  onChange={(e) => setPasswordCount(parseInt(e.target.value))}
                  className="w-full p-2 border rounded"
                >
                  {[1, 2, 3, 4, 5].map(count => (
                    <option key={count} value={count}>{count}个</option>
                  ))}
                </select>
              </div>
            </div>
          </div>
        </div>

        {/* Error Message */}
        {error && (
          <div className="p-2 bg-red-50 border border-red-200 rounded text-red-700">
            {error}
          </div>
        )}

        {/* Password Results */}
        <div className="space-y-4">
          <div className="flex justify-between items-center">
            <h3 className="text-lg font-semibold">生成的密码</h3>
            <div className="flex space-x-2">
              <Button onClick={() => setShowPassword(!showPassword)} variant="secondary" size="sm">
                {showPassword ? '隐藏' : '显示'}
              </Button>
              <Button onClick={handleRegenerate} variant="primary" size="sm">
                重新生成
              </Button>
            </div>
          </div>
          
          <div className="space-y-3">
            {passwords.map((password, index) => (
              <div key={index} className="flex justify-between items-center p-3 border border-border-secondary bg-bg-secondary rounded-lg">
                <div className="flex items-center space-x-2">
                  <span className="text-sm text-text-secondary">{index + 1}.</span>
                  <input
                    type={showPassword ? 'text' : 'password'}
                    value={password}
                    readOnly
                    className="bg-transparent border-none focus:outline-none"
                  />
                </div>
                <Button onClick={() => handleCopyPassword(password)} disabled={!password} variant="secondary" size="sm">
                  复制
                </Button>
              </div>
            ))}
          </div>
          
          {/* Password Strength */}
          {passwords.length > 0 && passwords[0] && (
            <div className="space-y-2">
              <div className="flex justify-between items-center">
                <span className="text-sm text-text-secondary">密码强度</span>
                <span className={`px-3 py-1 rounded-full text-sm font-medium ${getPasswordStrengthColor(passwordStrength)}`}>
                  {getPasswordStrengthText(passwordStrength)}
                </span>
              </div>
              <div className="w-full bg-bg-tertiary rounded-full h-2.5">
                <div
                  className={`h-2.5 rounded-full transition-all duration-300 ${getPasswordStrengthColor(passwordStrength).replace('text-', 'bg-').replace('-100', '-400')}`}
                  style={{ width: `${(passwordStrength === 'weak' ? 25 : passwordStrength === 'medium' ? 50 : passwordStrength === 'strong' ? 75 : 100)}%` }}
                ></div>
              </div>
            </div>
          )}
        </div>

        {/* Usage Instructions */}
        <UsageInstructions
          items={[
            '调整密码长度滑块设置所需的密码长度',
            '选择需要包含的字符类型（大写字母、小写字母、数字、特殊符号）',
            '设置高级选项（排除相似字符、排除特定字符、生成密码数量）',
            '查看生成的密码，使用"显示/隐藏"按钮切换密码可见性',
            '点击"复制"按钮复制密码，"重新生成"按钮生成新密码',
            '密码强度指示器会显示密码的安全等级'
          ]}
        />
      </div>
    )
  }
  ```

## 四、性能优化

### 3. 内存管理

- 避免内存泄漏
