# 闹钟工具技术设计文档

## 一、概述

本文档详细描述MaxTool 0.2版本中闹钟工具（Alarm Clock）的技术设计，包括功能设计、技术实现、组件结构和代码复用等方面。设计严格遵循项目规范，注重代码复用、组件化和性能优化。

## 二、功能设计

### 1. 核心功能

#### 闹钟设置
- 支持设置多个闹钟
- 选择小时和分钟
- 设置AM/PM或24小时制

#### 闹钟控制
- 开启/关闭闹钟
- 暂停闹钟（贪睡功能）
- 删除闹钟

#### 提醒方式
- 声音提醒（使用浏览器Web Audio API）
- 视觉提醒（闪烁效果）
- 通知提醒（使用浏览器Notification API）

### 2. 用户界面

#### 时钟显示
- 显示当前时间（模拟时钟或数字时钟）
- 显示当前日期

#### 闹钟列表
- 显示所有已设置的闹钟
- 每个闹钟显示时间、状态（开启/关闭）
- 提供编辑、删除、暂停按钮

#### 功能按钮
- 添加闹钟按钮
- 设置按钮（切换12/24小时制等）

#### 使用说明
- 提供简洁的使用步骤说明

## 三、技术设计

### 1. 技术选型

#### 核心库
- 原生JavaScript：用于时钟和闹钟的核心逻辑

#### 浏览器API
- Web Audio API：用于播放闹钟声音
- Notification API：用于发送通知提醒
- LocalStorage API：用于持久化存储闹钟设置

### 2. 组件结构

#### 主组件

##### AlarmClock
- **位置**：`components/tools/AlarmClock.tsx`
- **功能**：闹钟工具的主组件
- **结构**：
  ```tsx
  'use client'

  import { useState, useEffect, useRef } from 'react'
  import Button from '@/components/ui/Button'
  import Input from '@/components/ui/Input'
  import { trackUsage } from '@/lib/tracking'
  import UsageInstructions from '@/components/shared/UsageInstructions'

  // 定义闹钟数据结构
  interface Alarm {
    id: string;
    hour: number;
    minute: number;
    is24Hour: boolean;
    ampm: 'am' | 'pm';
    isEnabled: boolean;
    isRinging: boolean;
  }

  // 默认闹钟声音（简单的蜂鸣音）
  const DEFAULT_ALARM_SOUND = {
    frequency: 800,
    duration: 1000,
    interval: 2000
  };

  export default function AlarmClock() {
    // State declarations
    const [currentTime, setCurrentTime] = useState(new Date());
    const [alarms, setAlarms] = useState<Alarm[]>([]);
    const [isAddingAlarm, setIsAddingAlarm] = useState(false);
    const [newAlarm, setNewAlarm] = useState<Omit<Alarm, 'id' | 'isRinging'>>({
      hour: 8,
      minute: 0,
      is24Hour: true,
      ampm: 'am',
      isEnabled: true
    });
    const [showNotification, setShowNotification] = useState(false);
    const [notificationPermission, setNotificationPermission] = useState('default');
    const [hasTracked, setHasTracked] = useState(false);
    
    // Refs
    const audioContextRef = useRef<AudioContext | null>(null);
    const intervalRef = useRef<NodeJS.Timeout | null>(null);
    const alarmSoundRef = useRef<NodeJS.Timeout | null>(null);

    // 初始化
    useEffect(() => {
      // 请求通知权限
      if ('Notification' in window) {
        Notification.requestPermission().then(permission => {
          setNotificationPermission(permission);
        });
      }

      // 从localStorage加载闹钟设置
      const savedAlarms = localStorage.getItem('maxtool-alarms');
      if (savedAlarms) {
        try {
          setAlarms(JSON.parse(savedAlarms));
        } catch (error) {
          console.error('Failed to parse saved alarms:', error);
        }
      }

      // 设置当前时间更新定时器
      const timeInterval = setInterval(() => {
        setCurrentTime(new Date());
      }, 1000);

      // 跟踪工具使用
      trackUsage('alarm-clock');
      setHasTracked(true);

      return () => {
        clearInterval(timeInterval);
        if (intervalRef.current) clearInterval(intervalRef.current);
        if (alarmSoundRef.current) clearInterval(alarmSoundRef.current);
        if (audioContextRef.current) audioContextRef.current.close();
      };
    }, []);

    // 保存闹钟到localStorage
    useEffect(() => {
      localStorage.setItem('maxtool-alarms', JSON.stringify(alarms));
    }, [alarms]);

    // 检查闹钟是否触发
    useEffect(() => {
      const checkAlarms = () => {
        const now = new Date();
        const currentHour = now.getHours();
        const currentMinute = now.getMinutes();

        setAlarms(prevAlarms => {
          return prevAlarms.map(alarm => {
            let alarmHour = alarm.hour;
            
            // 转换12小时制到24小时制
            if (!alarm.is24Hour) {
              if (alarm.ampm === 'pm' && alarm.hour < 12) {
                alarmHour += 12;
              } else if (alarm.ampm === 'am' && alarm.hour === 12) {
                alarmHour = 0;
              }
            }

            // 检查是否到了闹钟时间
            if (alarm.isEnabled && !alarm.isRinging && alarmHour === currentHour && alarm.minute === currentMinute) {
              // 触发闹钟
              triggerAlarm();
              return { ...alarm, isRinging: true };
            }

            return alarm;
          });
        });
      };

      // 每分钟检查一次闹钟
      const alarmCheckInterval = setInterval(checkAlarms, 60000);
      
      return () => clearInterval(alarmCheckInterval);
    }, [currentTime]);

    // 播放闹钟声音
    const playAlarmSound = () => {
      if (!audioContextRef.current) {
        audioContextRef.current = new (window.AudioContext || (window as any).webkitAudioContext)();
      }

      const audioContext = audioContextRef.current;
      const oscillator = audioContext.createOscillator();
      const gainNode = audioContext.createGain();

      oscillator.connect(gainNode);
      gainNode.connect(audioContext.destination);

      oscillator.frequency.setValueAtTime(DEFAULT_ALARM_SOUND.frequency, audioContext.currentTime);
      oscillator.type = 'sine';

      // 音量渐变
      gainNode.gain.setValueAtTime(0, audioContext.currentTime);
      gainNode.gain.linearRampToValueAtTime(0.5, audioContext.currentTime + 0.1);
      gainNode.gain.linearRampToValueAtTime(0, audioContext.currentTime + DEFAULT_ALARM_SOUND.duration);

      oscillator.start(audioContext.currentTime);
      oscillator.stop(audioContext.currentTime + DEFAULT_ALARM_SOUND.duration);
    };

    // 触发闹钟
    const triggerAlarm = () => {
      // 播放声音
      playAlarmSound();
      
      // 设置声音循环
      if (alarmSoundRef.current) clearInterval(alarmSoundRef.current);
      alarmSoundRef.current = setInterval(() => {
        playAlarmSound();
      }, DEFAULT_ALARM_SOUND.interval);

      // 显示通知
      if (notificationPermission === 'granted' && 'Notification' in window) {
        new Notification('闹钟响了！', {
          body: '时间到了，请起床或处理任务',
          icon: '/favicon.ico'
        });
      }

      // 显示视觉提醒
      setShowNotification(true);
    };

    // 添加新闹钟
    const handleAddAlarm = () => {
      const newAlarmId = `alarm-${Date.now()}`;
      setAlarms(prevAlarms => [
        ...prevAlarms,
        { ...newAlarm, id: newAlarmId, isRinging: false }
      ]);
      setIsAddingAlarm(false);
      setNewAlarm({
        hour: 8,
        minute: 0,
        is24Hour: true,
        ampm: 'am',
        isEnabled: true
      });
    };

    // 切换闹钟状态
    const toggleAlarm = (alarmId: string) => {
      setAlarms(prevAlarms => {
        return prevAlarms.map(alarm => {
          if (alarm.id === alarmId) {
            return { ...alarm, isEnabled: !alarm.isEnabled };
          }
          return alarm;
        });
      });
    };

    // 暂停闹钟
    const snoozeAlarm = (alarmId: string) => {
      setAlarms(prevAlarms => {
        return prevAlarms.map(alarm => {
          if (alarm.id === alarmId) {
            // 重置闹钟状态
            return { ...alarm, isRinging: false };
          }
          return alarm;
        });
      });

      // 停止声音
      if (alarmSoundRef.current) {
        clearInterval(alarmSoundRef.current);
        alarmSoundRef.current = null;
      }

      setShowNotification(false);
    };

    // 删除闹钟
    const deleteAlarm = (alarmId: string) => {
      setAlarms(prevAlarms => {
        return prevAlarms.filter(alarm => alarm.id !== alarmId);
      });
    };

    // 格式化时间显示
    const formatTime = (hour: number, minute: number, is24Hour: boolean, ampm?: 'am' | 'pm') => {
      let displayHour = hour;
      let displayAmpm = '';

      if (!is24Hour) {
        displayAmpm = ampm || 'am';
        if (hour > 12) {
          displayHour = hour - 12;
          displayAmpm = 'pm';
        } else if (hour === 0) {
          displayHour = 12;
          displayAmpm = 'am';
        } else if (hour === 12) {
          displayAmpm = 'pm';
        }
      }

      const displayMinute = minute.toString().padStart(2, '0');
      return `${displayHour}:${displayMinute}${displayAmpm ? ` ${displayAmpm.toUpperCase()}` : ''}`;
    };

    // 获取当前时间显示
    const getCurrentTimeDisplay = () => {
      return formatTime(
        currentTime.getHours(),
        currentTime.getMinutes(),
        true
      );
    };

    // 获取当前日期显示
    const getCurrentDateDisplay = () => {
      const options: Intl.DateTimeFormatOptions = {
        year: 'numeric',
        month: 'long',
        day: 'numeric',
        weekday: 'long'
      };
      return currentTime.toLocaleDateString('zh-CN', options);
    };

    // Render
    return (
      <div className="space-y-6">
        {/* Clock Display */}
        <div className="space-y-4">
          <h2 className="text-xl font-semibold">闹钟</h2>
          
          <div className="text-center space-y-2">
            <div className="text-5xl font-bold">{getCurrentTimeDisplay()}</div>
            <div className="text-lg text-text-secondary">{getCurrentDateDisplay()}</div>
          </div>
        </div>

        {/* Alarm List */}
        <div className="space-y-4">
          <h3 className="text-lg font-semibold">我的闹钟</h3>
          
          {alarms.length === 0 ? (
            <div className="text-center p-8 bg-bg-secondary rounded-lg">
              <p className="text-text-secondary">还没有设置闹钟</p>
              <p className="text-sm text-text-tertiary">点击下方按钮添加第一个闹钟</p>
            </div>
          ) : (
            <div className="space-y-3">
              {alarms.map(alarm => (
                <div key={alarm.id} className={`p-4 rounded-lg border ${alarm.isRinging ? 'border-red-500 bg-red-50' : 'border-border-secondary bg-bg-secondary'}`}>
                  <div className="flex justify-between items-center">
                    <div className="flex items-center space-x-4">
                      <input
                        type="checkbox"
                        checked={alarm.isEnabled}
                        onChange={() => toggleAlarm(alarm.id)}
                        className="w-5 h-5"
                      />
                      <div className={`text-xl font-medium ${alarm.isRinging ? 'text-red-600' : ''}`}>
                        {formatTime(alarm.hour, alarm.minute, alarm.is24Hour, alarm.ampm)}
                      </div>
                    </div>
                    
                    <div className="flex space-x-2">
                      {alarm.isRinging && (
                        <Button onClick={() => snoozeAlarm(alarm.id)} variant="secondary" size="sm">
                          暂停
                        </Button>
                      )}
                      <Button onClick={() => deleteAlarm(alarm.id)} variant="danger" size="sm">
                        删除
                      </Button>
                    </div>
                  </div>
                </div>
              ))}
            </div>
          )}
        </div>

        {/* Add Alarm */}
        <div className="space-y-4">
          {isAddingAlarm ? (
            <div className="p-4 bg-bg-secondary rounded-lg space-y-4">
              <h3 className="text-lg font-semibold">添加新闹钟</h3>
              
              <div className="grid grid-cols-1 md:grid-cols-3 gap-4">
                {/* Hour Input */}
                <div className="space-y-2">
                  <label className="block">小时</label>
                  <Input
                    type="number"
                    value={newAlarm.hour}
                    onChange={(e) => setNewAlarm({ ...newAlarm, hour: parseInt(e.target.value) || 0 })}
                    min={newAlarm.is24Hour ? 0 : 1}
                    max={newAlarm.is24Hour ? 23 : 12}
                    className="w-full"
                  />
                </div>

                {/* Minute Input */}
                <div className="space-y-2">
                  <label className="block">分钟</label>
                  <Input
                    type="number"
                    value={newAlarm.minute}
                    onChange={(e) => setNewAlarm({ ...newAlarm, minute: parseInt(e.target.value) || 0 })}
                    min={0}
                    max={59}
                    className="w-full"
                  />
                </div>

                {/* AM/PM or 24 Hour */}
                <div className="space-y-2">
                  <label className="block">时间格式</label>
                  <div className="flex items-center space-x-4">
                    <div className="flex items-center">
                      <input
                        type="radio"
                        id="24hour"
                        name="timeFormat"
                        checked={newAlarm.is24Hour}
                        onChange={() => setNewAlarm({ ...newAlarm, is24Hour: true })}
                      />
                      <label htmlFor="24hour" className="ml-2">24小时制</label>
                    </div>
                    <div className="flex items-center">
                      <input
                        type="radio"
                        id="12hour"
                        name="timeFormat"
                        checked={!newAlarm.is24Hour}
                        onChange={() => setNewAlarm({ ...newAlarm, is24Hour: false })}
                      />
                      <label htmlFor="12hour" className="ml-2">12小时制</label>
                    </div>
                  </div>
                </div>
              </div>

              {/* AM/PM Selection */}
              {!newAlarm.is24Hour && (
                <div className="space-y-2">
                  <label className="block">AM/PM</label>
                  <div className="flex space-x-4">
                    <Button
                      variant={newAlarm.ampm === 'am' ? 'primary' : 'secondary'}
                      onClick={() => setNewAlarm({ ...newAlarm, ampm: 'am' })}
                      size="sm"
                    >
                      AM
                    </Button>
                    <Button
                      variant={newAlarm.ampm === 'pm' ? 'primary' : 'secondary'}
                      onClick={() => setNewAlarm({ ...newAlarm, ampm: 'pm' })}
                      size="sm"
                    >
                      PM
                    </Button>
                  </div>
                </div>
              )}

              <div className="flex justify-between">
                <Button onClick={() => setIsAddingAlarm(false)} variant="secondary">
                  取消
                </Button>
                <Button onClick={handleAddAlarm} variant="primary">
                  保存闹钟
                </Button>
              </div>
            </div>
          ) : (
            <Button onClick={() => setIsAddingAlarm(true)} variant="primary" className="w-full">
              添加新闹钟
            </Button>
          )}
        </div>

        {/* Visual Alarm Notification */}
        {showNotification && (
          <div className="fixed top-4 right-4 bg-red-500 text-white p-4 rounded-lg shadow-lg z-50 animate-pulse">
            <div className="flex items-center space-x-4">
              <div className="text-xl">⏰</div>
              <div>
                <div className="font-semibold">闹钟响了！</div>
                <div className="text-sm">时间到了，请起床或处理任务</div>
              </div>
              <Button onClick={() => setShowNotification(false)} variant="secondary" size="sm">
                关闭
              </Button>
            </div>
          </div>
        )}

        {/* Usage Instructions */}
        <UsageInstructions
          items={[
            '点击"添加新闹钟"按钮设置第一个闹钟',
            '选择小时和分钟，设置时间格式（12/24小时制）',
            '开启闹钟开关使其生效',
            '当闹钟响起时，点击"暂停"按钮停止提醒',
            '可以设置多个闹钟，满足不同时间需求'
          ]}
        />
      </div>
    )
  }
  ```
