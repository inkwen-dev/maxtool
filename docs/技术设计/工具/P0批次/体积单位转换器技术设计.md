# 体积单位转换器技术设计文档

## 一、概述

本文档详细描述MaxTool 0.2版本中体积单位转换器（Volume Unit Converter）工具的技术设计，包括功能设计、技术实现、组件结构和代码复用等方面。设计严格遵循项目规范，注重代码复用、组件化和性能优化。

## 二、功能设计

### 1. 核心功能

#### 基本单位转换
- 支持多种体积单位之间的相互转换
- 实时显示转换结果
- 支持输入任意数值

#### 单位分类
- 公制单位：立方米、立方分米、立方厘米、升等
- 英制单位：立方英尺、立方英寸、加仑等
- 美制单位：美制加仑等

### 2. 用户界面

#### 输入区域
- 数值输入框：用于输入要转换的数值
- 源单位选择：选择输入数值的单位

#### 输出区域
- 转换结果显示：以不同单位显示转换后的结果
- 常用单位快速访问：显示常用单位的转换结果

#### 功能按钮
- 交换单位按钮：快速交换源单位和目标单位
- 复制按钮：复制转换结果

#### 使用说明
- 提供简洁的使用步骤说明

### 2. 组件结构

#### 主组件

##### VolumeUnitConverter
- **位置**：`components/tools/VolumeUnitConverter.tsx`
- **功能**：体积单位转换器工具的主组件
- **结构**：
  ```tsx
  'use client'

  import { useState, useEffect } from 'react'
  import Button from '@/components/ui/Button'
  import Input from '@/components/ui/Input'
  import { trackUsage } from '@/lib/tracking'
  import UsageInstructions from '@/components/shared/UsageInstructions'
  import { copyToClipboard } from '@/lib/clipboard-utils'

  // 定义支持的体积单位
  const VOLUME_UNITS = [
    // 公制单位
    { id: 'm3', name: '立方米', symbol: 'm³', toCubicMeter: 1 },
    { id: 'dm3', name: '立方分米', symbol: 'dm³', toCubicMeter: 0.001 },
    { id: 'cm3', name: '立方厘米', symbol: 'cm³', toCubicMeter: 0.000001 },
    { id: 'mm3', name: '立方毫米', symbol: 'mm³', toCubicMeter: 0.000000001 },
    { id: 'l', name: '升', symbol: 'L', toCubicMeter: 0.001 },
    { id: 'ml', name: '毫升', symbol: 'mL', toCubicMeter: 0.000001 },
    
    // 英制单位
    { id: 'ft3', name: '立方英尺', symbol: 'ft³', toCubicMeter: 0.0283168466 },
    { id: 'in3', name: '立方英寸', symbol: 'in³', toCubicMeter: 0.000016387064 },
    { id: 'gal', name: '英制加仑', symbol: 'gal (UK)', toCubicMeter: 0.00454609 },
    { id: 'qt', name: '英制夸脱', symbol: 'qt (UK)', toCubicMeter: 0.0011365225 },
    { id: 'pt', name: '英制品脱', symbol: 'pt (UK)', toCubicMeter: 0.00056826125 },
    
    // 美制单位
    { id: 'gal_us', name: '美制加仑', symbol: 'gal (US)', toCubicMeter: 0.00378541178 },
    { id: 'qt_us', name: '美制夸脱', symbol: 'qt (US)', toCubicMeter: 0.000946352946 },
    { id: 'pt_us', name: '美制品脱', symbol: 'pt (US)', toCubicMeter: 0.000473176473 },
    { id: 'fl_oz_us', name: '美制液盎司', symbol: 'fl oz (US)', toCubicMeter: 0.0000295735296 }
  ] as const

  // 常用单位组（用于快速访问）
  const COMMON_UNITS = ['m3', 'l', 'ml', 'ft3', 'gal', 'gal_us', 'in3', 'cm3'] as const

  export default function VolumeUnitConverter() {
    // State declarations
    const [inputValue, setInputValue] = useState('1')
    const [fromUnit, setFromUnit] = useState('l')
    const [toUnit, setToUnit] = useState('gal')
    const [convertedValues, setConvertedValues] = useState<Record<string, number>>({})
    const [error, setError] = useState<string | null>(null)
    const [hasTracked, setHasTracked] = useState(false)

    // 转换逻辑
    const convertUnits = () => {
      const value = parseFloat(inputValue)
      
      if (isNaN(value)) {
        setError('请输入有效的数值')
        setConvertedValues({})
        return
      }

      try {
        // 获取源单位和目标单位的转换系数
        const fromUnitData = VOLUME_UNITS.find(u => u.id === fromUnit)
        const toUnitData = VOLUME_UNITS.find(u => u.id === toUnit)
        
        if (!fromUnitData || !toUnitData) {
          setError('无效的单位选择')
          setConvertedValues({})
          return
        }

        // 先转换为立方米，再转换为目标单位
        const valueInCubicMeters = value * fromUnitData.toCubicMeter
        const result = valueInCubicMeters / toUnitData.toCubicMeter

        // 计算所有单位的转换结果（用于显示常用单位）
        const allConvertedValues: Record<string, number> = {}
        VOLUME_UNITS.forEach(unit => {
          allConvertedValues[unit.id] = valueInCubicMeters / unit.toCubicMeter
        })

        setConvertedValues(allConvertedValues)
        setError(null)

        if (!hasTracked) {
          trackUsage('volume-unit-converter')
          setHasTracked(true)
        }
      } catch (err) {
        setError(err instanceof Error ? err.message : '转换失败')
        setConvertedValues({})
      }
    }

    // 输入变化时自动转换
    useEffect(() => {
      convertUnits()
    }, [inputValue, fromUnit, toUnit])

    // Event handlers
    const handleInputChange = (e: React.ChangeEvent<HTMLInputElement>) => {
      setInputValue(e.target.value)
    }

    const handleFromUnitChange = (e: React.ChangeEvent<HTMLSelectElement>) => {
      setFromUnit(e.target.value)
    }

    const handleToUnitChange = (e: React.ChangeEvent<HTMLSelectElement>) => {
      setToUnit(e.target.value)
    }

    const handleSwapUnits = () => {
      setFromUnit(toUnit)
      setToUnit(fromUnit)
    }

    const handleCopy = () => {
      if (convertedValues[toUnit] !== undefined) {
        const result = `${convertedValues[toUnit].toFixed(6)} ${toUnit}`
        copyToClipboard(result)
        // 显示复制成功提示
      }
    }

    // 格式化数值显示
    const formatValue = (value: number): string => {
      if (Math.abs(value) >= 1e6 || Math.abs(value) < 1e-3) {
        return value.toExponential(6)
      }
      return value.toFixed(6).replace(/\.?0+$/, '')
    }

    // Render
    return (
      <div className="space-y-6">
        {/* Input Section */}
        <div className="space-y-4">
          <h2 className="text-xl font-semibold">体积单位转换</h2>
          
          <div className="grid grid-cols-1 md:grid-cols-3 gap-4 items-end">
            {/* Input Value */}
            <div className="space-y-2">
              <label className="block">数值</label>
              <Input
                type="number"
                value={inputValue}
                onChange={handleInputChange}
                placeholder="输入数值"
                step="any"
                className="w-full"
              />
            </div>

            {/* From Unit */}
            <div className="space-y-2">
              <label className="block">从</label>
              <select
                value={fromUnit}
                onChange={handleFromUnitChange}
                className="w-full p-2 border rounded"
              >
                {VOLUME_UNITS.map(unit => (
                  <option key={unit.id} value={unit.id}>
                    {unit.name} ({unit.symbol})
                  </option>
                ))}
              </select>
            </div>

            {/* To Unit */}
            <div className="space-y-2">
              <div className="flex justify-between items-center">
                <label className="block">到</label>
                <Button onClick={handleSwapUnits} variant="secondary" size="sm">
                  交换
                </Button>
              </div>
              <select
                value={toUnit}
                onChange={handleToUnitChange}
                className="w-full p-2 border rounded"
              >
                {VOLUME_UNITS.map(unit => (
                  <option key={unit.id} value={unit.id}>
                    {unit.name} ({unit.symbol})
                  </option>
                ))}
              </select>
            </div>
          </div>

          {/* Error Message */}
          {error && (
            <div className="p-2 bg-red-50 border border-red-200 rounded text-red-700">
              {error}
            </div>
          )}
        </div>

        {/* Main Result */}
        {!error && convertedValues[toUnit] !== undefined && (
          <div className="space-y-4">
            <div className="text-center">
              <h3 className="text-lg font-medium mb-2">转换结果</h3>
              <div className="text-4xl font-bold">
                {formatValue(convertedValues[toUnit])} {VOLUME_UNITS.find(u => u.id === toUnit)?.symbol}
              </div>
              <div className="text-sm text-text-secondary mt-2">
                {inputValue} {VOLUME_UNITS.find(u => u.id === fromUnit)?.symbol} = {formatValue(convertedValues[toUnit])} {VOLUME_UNITS.find(u => u.id === toUnit)?.symbol}
              </div>
            </div>

            <Button onClick={handleCopy} disabled={!convertedValues[toUnit]} variant="secondary" className="w-full">
              复制结果
            </Button>
          </div>
        )}

        {/* Common Units Results */}
        {!error && Object.keys(convertedValues).length > 0 && (
          <div className="space-y-4">
            <h3 className="text-lg font-semibold">常用单位转换结果</h3>
            <div className="grid grid-cols-2 md:grid-cols-4 gap-4">
              {COMMON_UNITS.map(unitId => {
                const unitData = VOLUME_UNITS.find(u => u.id === unitId)
                if (!unitData || convertedValues[unitId] === undefined) return null
                
                return (
                  <div key={unitId} className="p-4 bg-bg-secondary rounded-lg">
                    <div className="text-sm text-text-secondary mb-1">{unitData.name}</div>
                    <div className="text-xl font-medium">{formatValue(convertedValues[unitId])}</div>
                    <div className="text-xs text-text-tertiary">{unitData.symbol}</div>
                  </div>
                )
              })}
            </div>
          </div>
        )}

        {/* All Units Results */}
        {!error && Object.keys(convertedValues).length > 0 && (
          <div className="space-y-4">
            <h3 className="text-lg font-semibold">所有单位转换结果</h3>
            <div className="overflow-x-auto">
              <table className="min-w-full border-collapse">
                <thead>
                  <tr className="bg-bg-secondary">
                    <th className="border p-2 text-left">单位</th>
                    <th className="border p-2 text-left">符号</th>
                    <th className="border p-2 text-right">转换结果</th>
                  </tr>
                </thead>
                <tbody>
                  {VOLUME_UNITS.map(unit => {
                    if (convertedValues[unit.id] === undefined) return null
                    
                    return (
                      <tr key={unit.id} className={COMMON_UNITS.includes(unit.id as any) ? 'bg-bg-primary' : ''}>
                        <td className="border p-2">{unit.name}</td>
                        <td className="border p-2">{unit.symbol}</td>
                        <td className="border p-2 text-right font-mono">{formatValue(convertedValues[unit.id])}</td>
                      </tr>
                    )
                  })}
                </tbody>
              </table>
            </div>
          </div>
        )}

        {/* Usage Instructions */}
        <UsageInstructions
          items={[
            '在数值输入框中输入要转换的体积数值',
            '选择源单位（要转换的单位）',
            '选择目标单位（转换后的单位）',
            '查看转换结果，支持自动实时转换',
            '使用"交换"按钮快速交换源单位和目标单位',
            '点击"复制结果"按钮复制转换结果'
          ]}
        />
      </div>
    )
  }
  ```

## 三、技术实现

### 1. 单位转换实现

使用基于立方米的转换系数进行体积单位转换，所有单位先转换为立方米，再转换为目标单位。

```typescript
export function convertVolume(value: number, fromUnit: string, toUnit: string): number {
  const fromUnitData = VOLUME_UNITS.find(u => u.id === fromUnit)
  const toUnitData = VOLUME_UNITS.find(u => u.id === toUnit)
  
  if (!fromUnitData || !toUnitData) {
    throw new Error('无效的单位选择')
  }
  
  // 先转换为立方米，再转换为目标单位
  const valueInCubicMeters = value * fromUnitData.toCubicMeter
  return valueInCubicMeters / toUnitData.toCubicMeter
}
```

### 2. 实时转换实现

使用React的useEffect钩子监听输入值和单位选择的变化，自动触发转换计算。

```typescript
useEffect(() => {
  convertUnits()
}, [inputValue, fromUnit, toUnit])
```

### 3. 数值格式化实现

根据数值的大小选择合适的格式化方式，对于非常大或非常小的数值使用科学计数法。

```typescript
export function formatNumber(value: number, precision: number = 6): string {
  if (Math.abs(value) >= 1e6 || Math.abs(value) < 1e-3) {
    return value.toExponential(precision)
  }
  return value.toFixed(precision).replace(/\.?0+$/, '')
}
```

### 4. 复制功能实现

使用共享的`copyToClipboard`函数将转换结果复制到剪贴板。

```typescript
const handleCopy = () => {

    copyToClipboard(result)
    // 显示复制成功提示
  }
}
```
