# 在线尺子技术设计文档

## 一、概述

本文档详细描述MaxTool 0.2版本中在线尺子（Online Ruler）的技术设计，包括功能设计、技术实现、组件结构和代码复用等方面。设计严格遵循项目规范，注重代码复用、组件化和性能优化。

## 二、功能设计

### 1. 核心功能

#### 基本尺子功能
- 提供水平尺子：用于测量水平方向的长度
- 提供垂直尺子：用于测量垂直方向的长度
- 支持公制单位（厘米、毫米）：默认使用公制单位
- 支持英制单位（英寸、分数）：提供英制单位切换
- 实时测量：显示当前鼠标位置对应的刻度值

#### 高级功能
- 放大/缩小功能：支持尺子的放大和缩小
- 校准功能：允许用户校准尺子的实际尺寸
- 锁定功能：锁定尺子的位置和大小
- 尺子样式自定义：支持不同的尺子样式（木质、金属、透明等）

### 2. 用户界面

#### 尺子显示区域
- 水平尺子：显示在页面顶部或自定义位置
- 垂直尺子：显示在页面左侧或自定义位置
- 测量标记：显示当前测量的起始和结束位置
- 测量结果显示：显示当前测量的长度值

#### 控制面板
- 单位切换按钮：切换公制/英制单位
- 放大/缩小按钮：调整尺子的比例
- 校准按钮：进入校准模式
- 锁定按钮：锁定/解锁尺子位置
- 样式选择：选择不同的尺子样式

#### 使用说明
- 提供简洁的使用步骤说明

## 三、技术设计

### 1. 技术选型

#### 核心库
- **原生JavaScript**：用于实现尺子的核心功能
- **React**：用于构建用户界面
- **react-draggable**：用于实现尺子的拖拽功能

#### 浏览器API
- Window.getComputedStyle：用于获取元素的实际尺寸
- ResizeObserver：用于监听窗口大小变化
- MouseEvent：用于处理鼠标事件和测量逻辑

### 2. 组件结构

#### 主组件

##### OnlineRuler
- **位置**：`components/tools/OnlineRuler.tsx`
- **功能**：在线尺子工具的主组件
- **结构**：
  ```tsx
  'use client'

  import { useState, useEffect, useRef } from 'react'
  import Draggable from 'react-draggable'
  import Button from '@/components/ui/Button'
  import { trackUsage } from '@/lib/tracking'
  import UsageInstructions from '@/components/shared/UsageInstructions'

  // 定义单位类型
  type UnitType = 'metric' | 'imperial'

  // 定义尺子样式类型
  type RulerStyle = 'wood' | 'metal' | 'transparent' | 'standard'

  // 定义测量状态
  interface Measurement {
    start: number | null
    end: number | null
    value: number | null
  }

  export default function OnlineRuler() {
    // State declarations
    const [unit, setUnit] = useState<UnitType>('metric')
    const [scale, setScale] = useState(1)
    const [isLocked, setIsLocked] = useState(false)
    const [style, setStyle] = useState<RulerStyle>('standard')
    const [isCalibrating, setIsCalibrating] = useState(false)
    const [calibrationValue, setCalibrationValue] = useState(1)
    const [horizontalMeasurement, setHorizontalMeasurement] = useState<Measurement>({
      start: null,
      end: null,
      value: null
    })
    const [verticalMeasurement, setVerticalMeasurement] = useState<Measurement>({
      start: null,
      end: null,
      value: null
    })
    const [isDragging, setIsDragging] = useState(false)
    const hasTracked = useRef(false)

    // 尺子容器引用
    const containerRef = useRef<HTMLDivElement>(null)
    const horizontalRulerRef = useRef<HTMLDivElement>(null)
    const verticalRulerRef = useRef<HTMLDivElement>(null)

    // 初始化
    useEffect(() => {
      if (!hasTracked.current) {
        trackUsage('online-ruler')
        hasTracked.current = true
      }

      // 校准尺子尺寸
      calibrateRuler()

      // 监听窗口大小变化
      const resizeObserver = new ResizeObserver(() => {
        calibrateRuler()
      })

      if (containerRef.current) {
        resizeObserver.observe(containerRef.current)
      }

      return () => {
        if (containerRef.current) {
          resizeObserver.unobserve(containerRef.current)
        }
      }
    }, [calibrationValue])

    // 校准尺子
    const calibrateRuler = () => {
      // 实现尺子校准逻辑
      console.log('Calibrating ruler...')
    }

    // 开始水平测量
    const startHorizontalMeasurement = (e: React.MouseEvent) => {
      if (isLocked) return
      
      const rect = horizontalRulerRef.current?.getBoundingClientRect()
      if (!rect) return
      
      const startPosition = e.clientX - rect.left
      setHorizontalMeasurement({
        start: startPosition,
        end: null,
        value: null
      })
      setIsDragging(true)
    }

    // 进行水平测量
    const measureHorizontal = (e: React.MouseEvent) => {
      if (isDragging && horizontalMeasurement.start !== null) {
        const rect = horizontalRulerRef.current?.getBoundingClientRect()
        if (!rect) return
        
        const endPosition = e.clientX - rect.left
        const distance = Math.abs(endPosition - horizontalMeasurement.start)
        const actualDistance = distance / (scale * calibrationValue)
        
        setHorizontalMeasurement(prev => ({
          ...prev,
          end: endPosition,
          value: actualDistance
        }))
      }
    }

    // 结束水平测量
    const endHorizontalMeasurement = () => {
      setIsDragging(false)
    }

    // 开始垂直测量
    const startVerticalMeasurement = (e: React.MouseEvent) => {
      if (isLocked) return
      
      const rect = verticalRulerRef.current?.getBoundingClientRect()
      if (!rect) return
      
      const startPosition = e.clientY - rect.top
      setVerticalMeasurement({
        start: startPosition,
        end: null,
        value: null
      })
      setIsDragging(true)
    }

    // 进行垂直测量
    const measureVertical = (e: React.MouseEvent) => {
      if (isDragging && verticalMeasurement.start !== null) {
        const rect = verticalRulerRef.current?.getBoundingClientRect()
        if (!rect) return
        
        const endPosition = e.clientY - rect.top
        const distance = Math.abs(endPosition - verticalMeasurement.start)
        const actualDistance = distance / (scale * calibrationValue)
        
        setVerticalMeasurement(prev => ({
          ...prev,
          end: endPosition,
          value: actualDistance
        }))
      }
    }

    // 结束垂直测量
    const endVerticalMeasurement = () => {
      setIsDragging(false)
    }

    // 切换单位
    const toggleUnit = () => {
      setUnit(prev => prev === 'metric' ? 'imperial' : 'metric')
    }

    // 放大
    const zoomIn = () => {
      setScale(prev => Math.min(prev + 0.1, 3))
    }

    // 缩小
    const zoomOut = () => {
      setScale(prev => Math.max(prev - 0.1, 0.5))
    }

    // 重置缩放
    const resetZoom = () => {
      setScale(1)
    }

    // 切换锁定状态
    const toggleLock = () => {
      setIsLocked(prev => !prev)
    }

    // 开始校准
    const startCalibration = () => {
      setIsCalibrating(true)
    }

    // 保存校准
    const saveCalibration = () => {
      setIsCalibrating(false)
      // 实现校准保存逻辑
    }

    // 取消校准
    const cancelCalibration = () => {
      setIsCalibrating(false)
      setCalibrationValue(1)
    }

    // 格式化测量结果
    const formatMeasurement = (value: number | null): string => {
      if (value === null) return ''
      
      if (unit === 'metric') {
        // 公制单位：厘米和毫米
        const cm = Math.floor(value / 10)
        const mm = Math.round(value % 10)
        return `${cm}.${mm} cm (${Math.round(value)} mm)`
      } else {
        // 英制单位：英寸和分数
        const inches = value / 25.4
        const wholeInches = Math.floor(inches)
        const fraction = inches - wholeInches
        
        // 转换为最接近的分数（1/16英寸）
        const sixteenths = Math.round(fraction * 16)
        const reducedFraction = reduceFraction(sixteenths, 16)
        
        if (wholeInches === 0) {
          return `${reducedFraction.numerator}/${reducedFraction.denominator}" (${inches.toFixed(3)}")`
        } else if (reducedFraction.numerator === 0) {
          return `${wholeInches}" (${inches.toFixed(3)}")`
        } else {
          return `${wholeInches} ${reducedFraction.numerator}/${reducedFraction.denominator}" (${inches.toFixed(3)}")`
        }
      }
    }

    // 约分分数
    const reduceFraction = (numerator: number, denominator: number): { numerator: number; denominator: number } => {
      const gcd = (a: number, b: number): number => {
        return b === 0 ? a : gcd(b, a % b)
      }
      
      const greatestCommonDivisor = gcd(numerator, denominator)
      return {
        numerator: numerator / greatestCommonDivisor,
        denominator: denominator / greatestCommonDivisor
      }
    }

    // 计算刻度位置
    const calculateTickPositions = (length: number, unit: UnitType, scale: number): number[] => {
      // 根据单位和缩放计算刻度位置
      const ticks: number[] = []
      const majorInterval = unit === 'metric' ? 10 : 25.4 // 1cm或1英寸
      const minorInterval = majorInterval / 10
      
      for (let i = 0; i <= length; i += minorInterval * scale) {
        ticks.push(i)
      }
      
      return ticks
    }

    // 绘制刻度
    const renderTicks = (length: number, orientation: 'horizontal' | 'vertical') => {
      const ticks = calculateTickPositions(length, unit, scale)
      return (
        <div className="relative w-full h-full">
          {ticks.map((tick, index) => {
            const isMajorTick = index % 10 === 0
            const tickLength = isMajorTick ? (orientation === 'horizontal' ? 15 : 30) : (orientation === 'horizontal' ? 8 : 15)
            const tickValue = (index / 10) * (unit === 'metric' ? 1 : 2.54)
            
            return (
              <div key={index} className="absolute">
                <div
                  className={`absolute ${isMajorTick ? 'bg-black' : 'bg-gray-500'}`}
                  style={{
                    left: orientation === 'horizontal' ? `${tick}px` : '0',
                    top: orientation === 'vertical' ? `${tick}px` : '0',
                    width: orientation === 'horizontal' ? '1px' : `${tickLength}px`,
                    height: orientation === 'horizontal' ? `${tickLength}px` : '1px',
                  }}
                />
                {isMajorTick && (
                  <div
                    className={`absolute text-xs font-bold ${orientation === 'horizontal' ? '-translate-x-1/2 mt-2' : '-translate-y-1/2 ml-2'}`}
                    style={{
                      left: orientation === 'horizontal' ? `${tick}px` : `${tickLength + 2}px`,
                      top: orientation === 'vertical' ? `${tick}px` : `${tickLength + 2}px`,
                    }}
                  >
                    {unit === 'metric' ? `${Math.round(tickValue)}` : `${(tickValue / 2.54).toFixed(1)}`}
                  </div>
                )}
              </div>
            )
          })}
        </div>
      )
    }

    // 渲染测量标记
    const renderMeasurementMarks = (measurement: Measurement, orientation: 'horizontal' | 'vertical') => {
      if (measurement.start === null) return null
      
      return (
        <div className="absolute inset-0 pointer-events-none">
          {/* 起始标记 */}
          <div
            className="absolute bg-red-500"
            style={{
              left: orientation === 'horizontal' ? `${measurement.start}px` : '0',
              top: orientation === 'vertical' ? `${measurement.start}px` : '0',
              width: orientation === 'horizontal' ? '2px' : '100%',
              height: orientation === 'vertical' ? '2px' : '100%',
            }}
          />
          
          {/* 结束标记 */}
          {measurement.end !== null && (
            <div
              className="absolute bg-blue-500"
              style={{
                left: orientation === 'horizontal' ? `${measurement.end}px` : '0',
                top: orientation === 'vertical' ? `${measurement.end}px` : '0',
                width: orientation === 'horizontal' ? '2px' : '100%',
                height: orientation === 'vertical' ? '2px' : '100%',
              }}
            />
          )}
          
          {/* 测量区域 */}
          {measurement.start !== null && measurement.end !== null && (
            <div
              className="absolute bg-yellow-200 opacity-50"
              style={{
                left: orientation === 'horizontal' ? `${Math.min(measurement.start, measurement.end)}px` : '0',
                top: orientation === 'vertical' ? `${Math.min(measurement.start, measurement.end)}px` : '0',
                width: orientation === 'horizontal' ? `${Math.abs(measurement.end - measurement.start)}px` : '100%',
                height: orientation === 'vertical' ? `${Math.abs(measurement.end - measurement.start)}px` : '100%',
              }}
            />
          )}
        </div>
      )
    }

    // Render
    return (
      <div ref={containerRef} className="space-y-6">
        {/* Header */}
        <div>
          <h2 className="text-xl font-semibold">在线尺子</h2>
        </div>

        {/* Control Panel */}
        <div className="p-4 bg-bg-secondary rounded-lg space-y-4">
          <div className="grid grid-cols-2 md:grid-cols-4 gap-4">
            {/* Unit Toggle */}
            <Button
              onClick={toggleUnit}
              variant="primary"
              className="w-full"
            >
              {unit === 'metric' ? '切换到英制' : '切换到公制'}
            </Button>

            {/* Zoom Controls */}
            <div className="flex space-x-2">
              <Button onClick={zoomIn} variant="secondary" className="flex-1">
                放大
              </Button>
              <Button onClick={zoomOut} variant="secondary" className="flex-1">
                缩小
              </Button>
              <Button onClick={resetZoom} variant="secondary" className="flex-1">
                重置
              </Button>
            </div>

            {/* Lock Toggle */}
            <Button
              onClick={toggleLock}
              variant={isLocked ? 'danger' : 'secondary'}
              className="w-full"
            >
              {isLocked ? '解锁' : '锁定'}
            </Button>

            {/* Calibration Button */}
            {isCalibrating ? (
              <div className="flex space-x-2">
                <Button onClick={saveCalibration} variant="primary" className="flex-1">
                  保存校准
                </Button>
                <Button onClick={cancelCalibration} variant="secondary" className="flex-1">
                  取消
                </Button>
              </div>
            ) : (
              <Button onClick={startCalibration} variant="secondary" className="w-full">
                校准
              </Button>
            )}
          </div>

          {/* Calibration Input */}
          {isCalibrating && (
            <div className="flex items-center space-x-2">
              <label className="text-sm">校准值：</label>
              <input
                type="number"
                value={calibrationValue}
                onChange={(e) => setCalibrationValue(parseFloat(e.target.value) || 1)}
                min="0.5"
                max="2"
                step="0.1"
                className="w-full p-2 border rounded"
              />
              <span className="text-sm">调整此值使尺子与实际物体匹配</span>
            </div>
          )}

          {/* Style Selection */}
          <div className="flex items-center space-x-2">
            <label className="text-sm">尺子样式：</label>
            <select
              value={style}
              onChange={(e) => setStyle(e.target.value as RulerStyle)}
              className="w-full p-2 border rounded"
            >
              <option value="standard">标准</option>
              <option value="wood">木质</option>
              <option value="metal">金属</option>
              <option value="transparent">透明</option>
            </select>
          </div>
        </div>

        {/* Ruler Display */}
        <div className="relative border border-gray-300 rounded-lg overflow-hidden bg-white min-h-[400px]">
          {/* Horizontal Ruler */}
          <Draggable
            disabled={isLocked}
            axis="x"
            bounds="parent"
          >
            <div
              ref={horizontalRulerRef}
              className={`h-16 border-b border-gray-300 ${getRulerStyleClass(style)} cursor-pointer`}
              onMouseDown={startHorizontalMeasurement}
              onMouseMove={measureHorizontal}
              onMouseUp={endHorizontalMeasurement}
              onMouseLeave={endHorizontalMeasurement}
            >
              {renderTicks(800, 'horizontal')}
              {renderMeasurementMarks(horizontalMeasurement, 'horizontal')}
            </div>
          </Draggable>

          {/* Vertical Ruler */}
          <Draggable
            disabled={isLocked}
            axis="y"
            bounds="parent"
          >
            <div
              ref={verticalRulerRef}
              className={`w-16 border-r border-gray-300 ${getRulerStyleClass(style)} cursor-pointer`}
              onMouseDown={startVerticalMeasurement}
              onMouseMove={measureVertical}
              onMouseUp={endVerticalMeasurement}
              onMouseLeave={endVerticalMeasurement}
            >
              {renderTicks(400, 'vertical')}
              {renderMeasurementMarks(verticalMeasurement, 'vertical')}
            </div>
          </Draggable>

          {/* Measurement Result */}
          <div className="absolute bottom-2 right-2 bg-white bg-opacity-80 p-2 rounded shadow">
            <div className="text-sm">
              <div>水平测量：{formatMeasurement(horizontalMeasurement.value)}</div>
              <div>垂直测量：{formatMeasurement(verticalMeasurement.value)}</div>
            </div>
          </div>
        </div>

        {/* Usage Instructions */}
        <UsageInstructions
          items={[
            '点击并拖动水平或垂直尺子进行测量',
            '使用单位切换按钮在公制和英制单位之间切换',
            '通过放大/缩小按钮调整尺子的比例',
            '使用校准功能确保尺子的准确性',
            '锁定功能可以固定尺子的位置',
            '选择不同的尺子样式以满足个人偏好'
          ]}
        />
      </div>
    )
  }

  // 辅助函数：获取尺子样式类
  function getRulerStyleClass(style: RulerStyle): string {
    switch (style) {
      case 'wood':
        return 'bg-[linear-gradient(to_right,#8B4513_0%,#A0522D_100%)]'
      case 'metal':
        return 'bg-[linear-gradient(to_right,#C0C0C0_0%,#A9A9A9_100%)]'
      case 'transparent':
        return 'bg-transparent'
      case 'standard':
      default:
        return 'bg-gray-100'
    }
  }

  // 辅助函数：约分分数
  function reduceFraction(numerator: number, denominator: number): { numerator: number; denominator: number } {
    const gcd = (a: number, b: number): number => {
      return b === 0 ? a : gcd(b, a % b)
    }
    
    const greatestCommonDivisor = gcd(numerator, denominator)
    return {
      numerator: numerator / greatestCommonDivisor,
      denominator: denominator / greatestCommonDivisor
    }
  }

  setHorizontalMeasurement({
    start: startPosition,
    end: null,
    value: null
  })
  setIsDragging(true)
}

// 进行水平测量
const measureHorizontal = (e: React.MouseEvent) => {
  if (isDragging && horizontalMeasurement.start !== null) {
    const rect = horizontalRulerRef.current?.getBoundingClientRect()
    if (!rect) return
    
    const endPosition = e.clientX - rect.left
    const distance = Math.abs(endPosition - horizontalMeasurement.start)
    const actualDistance = distance / (scale * calibrationValue)
    
    setHorizontalMeasurement(prev => ({
      ...prev,
      end: endPosition,
      value: actualDistance
    }))
  }
}

// 结束水平测量
const endHorizontalMeasurement = () => {
  setIsDragging(false)
}
```

### 3. 单位转换实现

实现公制和英制单位之间的转换：

```typescript
// 格式化测量结果
const formatMeasurement = (value: number | null): string => {
  if (value === null) return ''
  
  if (unit === 'metric') {
    // 公制单位：厘米和毫米
    const cm = Math.floor(value / 10)
    const mm = Math.round(value % 10)
    return `${cm}.${mm} cm (${Math.round(value)} mm)`
  } else {
    // 英制单位：英寸和分数
    const inches = value / 25.4
    const wholeInches = Math.floor(inches)
    const fraction = inches - wholeInches
    
    // 转换为最接近的分数（1/16英寸）
    const sixteenths = Math.round(fraction * 16)
    const reducedFraction = reduceFraction(sixteenths, 16)
    
    if (wholeInches === 0) {
      return `${reducedFraction.numerator}/${reducedFraction.denominator}" (${inches.toFixed(3)}")`
    } else if (reducedFraction.numerator === 0) {
      return `${wholeInches}" (${inches.toFixed(3)}")`
    } else {
      return `${wholeInches} ${reducedFraction.numerator}/${reducedFraction.denominator}" (${inches.toFixed(3)}")`
    }
  }
}
```
