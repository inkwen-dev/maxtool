# 日期计算器技术设计文档

## 一、概述

本文档详细描述MaxTool 0.2版本中日期计算器（Date Calculator）工具的技术设计，包括功能设计、技术实现、组件结构和代码复用等方面。设计严格遵循项目规范，注重代码复用、组件化和性能优化。

## 二、功能设计

### 1. 核心功能

#### 日期差值计算
- 计算两个日期之间的差值
- 支持多种时间单位显示（年、月、日、小时、分钟、秒）

#### 日期增减计算
- 在指定日期基础上增加或减少指定时间量
- 支持年、月、日、小时、分钟、秒等时间单位

#### 工作日计算
- 计算两个日期之间的工作日数量（排除周末和法定节假日）
- 支持自定义工作日（如调整周末为周六和周日以外的日期）

### 2. 用户界面

#### 日期计算模式选择
- 提供选项卡或切换按钮，选择不同的计算模式：
  - 日期差值计算
  - 日期增减计算
  - 工作日计算

#### 日期输入区域
- 日期选择器：用于选择起始日期和结束日期
- 时间输入：可选，用于精确到时分秒的计算
- 增减时间量输入：用于日期增减计算

#### 计算结果区域
- 显示计算结果，支持多种时间单位格式
- 提供复制结果功能

#### 使用说明
- 提供简洁的使用步骤说明

## 三、技术设计

### 1. 技术选型

#### 浏览器API

### 2. 组件结构

#### 主组件

##### DateCalculator
- **位置**：`components/tools/DateCalculator.tsx`
- **功能**：日期计算器工具的主组件
- **结构**：
  ```tsx
  'use client'

  import { useState } from 'react'
  import Button from '@/components/ui/Button'
  import Input from '@/components/ui/Input'
  import { trackUsage } from '@/lib/tracking'
  import UsageInstructions from '@/components/shared/UsageInstructions'
  import { copyToClipboard } from '@/lib/clipboard-utils'

  // 定义计算模式
  type CalculationMode = 'difference' | 'add-subtract' | 'business-days'

  // 定义时间单位
  type TimeUnit = 'years' | 'months' | 'days' | 'hours' | 'minutes' | 'seconds'

  // 定义结果格式
  interface DateDifferenceResult {
    totalMilliseconds: number
    years: number
    months: number
    days: number
    hours: number
    minutes: number
    seconds: number
  }

  export default function DateCalculator() {
    // State declarations
    const [mode, setMode] = useState<CalculationMode>('difference')
    const [startDate, setStartDate] = useState<string>(new Date().toISOString().split('T')[0])
    const [startTime, setStartTime] = useState<string>('00:00:00')
    const [endDate, setEndDate] = useState<string>(new Date().toISOString().split('T')[0])
    const [endTime, setEndTime] = useState<string>('00:00:00')
    const [addSubtractAmount, setAddSubtractAmount] = useState<number>(1)
    const [addSubtractUnit, setAddSubtractUnit] = useState<TimeUnit>('days')
    const [includeStartTime, setIncludeStartTime] = useState<boolean>(false)
    const [excludeWeekends, setExcludeWeekends] = useState<boolean>(true)
    const [result, setResult] = useState<string>('')
    const [resultDetails, setResultDetails] = useState<DateDifferenceResult | null>(null)
    const [error, setError] = useState<string | null>(null)
    const [hasTracked, setHasTracked] = useState(false)

    // 初始化
    useState(() => {
      trackUsage('date-calculator')
      setHasTracked(true)
    }, [])

    // 计算日期差值
    const calculateDateDifference = () => {
      try {
        const start = new Date(`${startDate}T${startTime}`)
        const end = new Date(`${endDate}T${endTime}`)

        if (isNaN(start.getTime()) || isNaN(end.getTime())) {
          throw new Error('请输入有效的日期和时间')
        }

        const diffMs = Math.abs(end.getTime() - start.getTime())
        const result = calculateTimeDifference(diffMs)

        setResultDetails(result)
        setResult(formatResult(result))
        setError(null)
      } catch (err) {
        setError(err instanceof Error ? err.message : '计算失败')
        setResult('')
        setResultDetails(null)
      }
    }

    // 计算日期增减
    const calculateDateAddSubtract = () => {
      try {
        const baseDate = new Date(`${startDate}T${startTime}`)

        if (isNaN(baseDate.getTime())) {
          throw new Error('请输入有效的日期和时间')
        }

        const newDate = addTimeToDate(baseDate, addSubtractAmount, addSubtractUnit)
        const formattedDate = formatDate(newDate)

        setResult(formattedDate)
        setResultDetails(null)
        setError(null)
      } catch (err) {
        setError(err instanceof Error ? err.message : '计算失败')
        setResult('')
        setResultDetails(null)
      }
    }

    // 计算工作日
    const calculateBusinessDays = () => {
      try {
        const start = new Date(`${startDate}T${startTime}`)
        const end = new Date(`${endDate}T${endTime}`)

        if (isNaN(start.getTime()) || isNaN(end.getTime())) {
          throw new Error('请输入有效的日期和时间')
        }

        const businessDays = countBusinessDays(start, end, excludeWeekends)
        setResult(`${businessDays} 个工作日`)
        setResultDetails(null)
        setError(null)
      } catch (err) {
        setError(err instanceof Error ? err.message : '计算失败')
        setResult('')
        setResultDetails(null)
      }
    }

    // 执行计算
    const handleCalculate = () => {
      setError(null)
      
      switch (mode) {
        case 'difference':
          calculateDateDifference()
          break
        case 'add-subtract':
          calculateDateAddSubtract()
          break
        case 'business-days':
          calculateBusinessDays()
          break
      }
    }

    // 计算时间差值
    const calculateTimeDifference = (ms: number): DateDifferenceResult => {
      const seconds = Math.floor(ms / 1000)
      const minutes = Math.floor(seconds / 60)
      const hours = Math.floor(minutes / 60)
      const days = Math.floor(hours / 24)
      const months = Math.floor(days / 30) // 近似计算
      const years = Math.floor(days / 365) // 近似计算

      return {
        totalMilliseconds: ms,
        years,
        months: months % 12,
        days: days % 30,
        hours: hours % 24,
        minutes: minutes % 60,
        seconds: seconds % 60
      }
    }

    // 日期增减
    const addTimeToDate = (date: Date, amount: number, unit: TimeUnit): Date => {
      const newDate = new Date(date)

      switch (unit) {
        case 'years':
          newDate.setFullYear(newDate.getFullYear() + amount)
          break
        case 'months':
          newDate.setMonth(newDate.getMonth() + amount)
          break
        case 'days':
          newDate.setDate(newDate.getDate() + amount)
          break
        case 'hours':
          newDate.setHours(newDate.getHours() + amount)
          break
        case 'minutes':
          newDate.setMinutes(newDate.getMinutes() + amount)
          break
        case 'seconds':
          newDate.setSeconds(newDate.getSeconds() + amount)
          break
      }

      return newDate
    }

    // 计算工作日数量
    const countBusinessDays = (start: Date, end: Date, excludeWeekends: boolean): number => {
      const startDate = new Date(start)
      const endDate = new Date(end)
      const isForward = endDate >= startDate
      let count = 0

      // 确保我们从较小的日期开始
      const current = new Date(isForward ? startDate : endDate)
      const target = new Date(isForward ? endDate : startDate)

      // 循环计算工作日
      while (current <= target) {
        const dayOfWeek = current.getDay()
        if (!excludeWeekends || (dayOfWeek !== 0 && dayOfWeek !== 6)) {
          count++
        }
        current.setDate(current.getDate() + 1)
      }

      // 如果是反向计算，保持结果为正数
      return count
    }

    // 格式化日期
    const formatDate = (date: Date): string => {
      return date.toLocaleString('zh-CN', {
        year: 'numeric',
        month: '2-digit',
        day: '2-digit',
        hour: '2-digit',
        minute: '2-digit',
        second: '2-digit'
      })
    }

    // 格式化结果
    const formatResult = (result: DateDifferenceResult): string => {
      const parts = []
      
      if (result.years > 0) parts.push(`${result.years}年`)
      if (result.months > 0) parts.push(`${result.months}月`)
      if (result.days > 0) parts.push(`${result.days}天`)
      if (result.hours > 0) parts.push(`${result.hours}小时`)
      if (result.minutes > 0) parts.push(`${result.minutes}分钟`)
      if (result.seconds > 0) parts.push(`${result.seconds}秒`)
      
      return parts.length > 0 ? parts.join(' ') : '0秒'
    }

    // 复制结果
    const handleCopyResult = () => {
      if (result) {
        copyToClipboard(result)
        // 显示复制成功提示
      }
    }

    // Event handlers
    const handleModeChange = (newMode: CalculationMode) => {
      setMode(newMode)
      setResult('')
      setResultDetails(null)
      setError(null)
    }

    // Render
    return (
      <div className="space-y-6">
        {/* Header */}
        <div className="space-y-4">
          <h2 className="text-xl font-semibold">日期计算器</h2>
        </div>

        {/* Mode Selection */}
        <div className="space-y-4">
          <div className="flex border-b">
            <button
              className={`px-5 py-2 border-b-2 font-medium ${mode === 'difference' ? 'border-primary text-primary' : 'border-transparent text-text-secondary hover:text-text-primary'}`}
              onClick={() => handleModeChange('difference')}
            >
              日期差值
            </button>
            <button
              className={`px-5 py-2 border-b-2 font-medium ${mode === 'add-subtract' ? 'border-primary text-primary' : 'border-transparent text-text-secondary hover:text-text-primary'}`}
              onClick={() => handleModeChange('add-subtract')}
            >
              日期增减
            </button>
            <button
              className={`px-5 py-2 border-b-2 font-medium ${mode === 'business-days' ? 'border-primary text-primary' : 'border-transparent text-text-secondary hover:text-text-primary'}`}
              onClick={() => handleModeChange('business-days')}
            >
              工作日计算
            </button>
          </div>
        </div>

        {/* Date Inputs */}
        <div className="space-y-6">
          {/* Common Date Inputs */}
          <div className="space-y-4">
            {mode === 'difference' || mode === 'business-days' ? (
              <>
                <div className="space-y-2">
                  <label htmlFor="startDate">起始日期</label>
                  <div className="flex space-x-2">
                    <Input
                      type="date"
                      id="startDate"
                      value={startDate}
                      onChange={(e) => setStartDate(e.target.value)}
                      className="flex-1"
                    />
                    {includeStartTime && (
                      <Input
                        type="time"
                        id="startTime"
                        value={startTime}
                        onChange={(e) => setStartTime(e.target.value)}
                      />
                    )}
                  </div>
                </div>
                
                <div className="space-y-2">
                  <label htmlFor="endDate">结束日期</label>
                  <div className="flex space-x-2">
                    <Input
                      type="date"
                      id="endDate"
                      value={endDate}
                      onChange={(e) => setEndDate(e.target.value)}
                      className="flex-1"
                    />
                    {includeStartTime && (
                      <Input
                        type="time"
                        id="endTime"
                        value={endTime}
                        onChange={(e) => setEndTime(e.target.value)}
                      />
                    )}
                  </div>
                </div>
              </>
            ) : (
              <div className="space-y-2">
                <label htmlFor="baseDate">基准日期</label>
                <div className="flex space-x-2">
                  <Input
                    type="date"
                    id="baseDate"
                    value={startDate}
                    onChange={(e) => setStartDate(e.target.value)}
                    className="flex-1"
                  />
                  {includeStartTime && (
                    <Input
                      type="time"
                      id="baseTime"
                      value={startTime}
                      onChange={(e) => setStartTime(e.target.value)}
                    />
                  )}
                </div>
              </div>
            )}
            
            {/* Include Time Option */}
            <div className="flex items-center space-x-2">
              <input
                type="checkbox"
                id="includeTime"
                checked={includeStartTime}
                onChange={(e) => setIncludeStartTime(e.target.checked)}
              />
              <label htmlFor="includeTime">包含时间</label>
            </div>
            
            {/* Add/Subtract Inputs */}
            {mode === 'add-subtract' && (
              <div className="grid grid-cols-2 gap-4">
                <div className="space-y-2">
                  <label htmlFor="amount">数量</label>
                  <Input
                    type="number"
                    id="amount"
                    value={addSubtractAmount}
                    onChange={(e) => setAddSubtractAmount(parseInt(e.target.value) || 0)}
                    min="-99999"
                    max="99999"
                  />
                </div>
                <div className="space-y-2">
                  <label htmlFor="unit">单位</label>
                  <select
                    id="unit"
                    value={addSubtractUnit}
                    onChange={(e) => setAddSubtractUnit(e.target.value as TimeUnit)}
                    className="w-full p-2 border rounded"
                  >
                    <option value="years">年</option>
                    <option value="months">月</option>
                    <option value="days">日</option>
                    <option value="hours">小时</option>
                    <option value="minutes">分钟</option>
                    <option value="seconds">秒</option>
                  </select>
                </div>
              </div>
            )}
            
            {/* Business Days Options */}
            {mode === 'business-days' && (
              <div className="flex items-center space-x-2">
                <input
                  type="checkbox"
                  id="excludeWeekends"
                  checked={excludeWeekends}
                  onChange={(e) => setExcludeWeekends(e.target.checked)}
                />
                <label htmlFor="excludeWeekends">排除周末</label>
              </div>
            )}
          </div>
          
          {/* Calculate Button */}
          <div className="flex justify-center">
            <Button onClick={handleCalculate} variant="primary" size="lg">
              计算
            </Button>
          </div>
        </div>

        {/* Error Message */}
        {error && (
          <div className="p-2 bg-red-50 border border-red-200 rounded text-red-700">
            {error}
          </div>
        )}

        {/* Result Display */}
        {result && (
          <div className="space-y-4">
            <h3 className="text-lg font-semibold">计算结果</h3>
            
            <div className="p-4 border border-border-secondary bg-bg-secondary rounded-lg">
              <div className="flex justify-between items-center">
                <div className="text-lg font-medium">{result}</div>
                <Button onClick={handleCopyResult} variant="secondary" size="sm">
                  复制
                </Button>
              </div>
              
              {/* Detailed Result */}
              {resultDetails && (
                <div className="mt-4 space-y-2">
                  <h4 className="text-sm font-semibold text-text-secondary">详细信息</h4>
                  <div className="grid grid-cols-2 gap-4 text-sm">
                    <div className="flex justify-between">
  ```
