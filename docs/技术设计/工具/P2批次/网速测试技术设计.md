# 网速测试技术设计文档

## 1. 概述

网速测试工具是一款帮助用户测量网络连接速度的在线网络工具。用户可以测试下载速度、上传速度和网络延迟，并查看测试结果的详细信息和历史记录。

## 2. 功能设计

### 2.1 核心功能
- 下载速度测试
- 上传速度测试
- 网络延迟测试（Ping）
- 抖动测试
- 测试服务器选择
- 测试结果详细信息显示（平均速度、最大速度、延迟等）
- 测试结果历史记录
- 测试结果分享功能
- 网络连接类型识别

### 2.2 辅助功能
- 深色/浅色模式支持
- 响应式设计，适配不同设备
- 使用说明和示例
- 网络速度相关知识参考

### 3.2 核心实现逻辑
```javascript
// 网速测试核心函数
const speedTest = {
  // 测量延迟（Ping）
  async measurePing(serverUrl) {
    const startTime = performance.now()
    
    try {
      await fetch(serverUrl, {
        method: 'HEAD',
        cache: 'no-cache'
      })
      const endTime = performance.now()
      return Math.round(endTime - startTime)
    } catch (error) {
      console.error('Ping测试失败:', error)
      throw new Error('Ping测试失败')
    }
  },
  
  // 测量下载速度
  async measureDownloadSpeed(serverUrl, fileSize = 10) {
    const startTime = performance.now()
    const response = await fetch(serverUrl, {
      cache: 'no-cache'
    })
    
    // 创建读取器
    const reader = response.body.getReader()
    let receivedBytes = 0
    
    try {
      while (true) {
        const { done, value } = await reader.read()
        if (done) break
        receivedBytes += value.length
      }
      
      const endTime = performance.now()
      const duration = (endTime - startTime) / 1000 // 秒
      const speedMbps = (receivedBytes * 8) / (1024 * 1024 * duration) // Mbps
      
      return {
        speed: Math.round(speedMbps * 10) / 10,
        bytes: receivedBytes,
        duration
      }
    } catch (error) {
      console.error('下载速度测试失败:', error)
      throw new Error('下载速度测试失败')
    } finally {
      reader.releaseLock()
    }
  },
  
  // 测量上传速度
  async measureUploadSpeed(serverUrl, fileSize = 10) {
    // 创建随机数据
    const data = new Uint8Array(fileSize * 1024 * 1024)
    crypto.getRandomValues(data)
    
    const startTime = performance.now()
    
    try {
      const response = await fetch(serverUrl, {
        method: 'POST',
        body: data,
        headers: {
          'Content-Type': 'application/octet-stream'
        }
      })
      
      if (!response.ok) {
        throw new Error('上传请求失败')
      }
      
      const endTime = performance.now()
      const duration = (endTime - startTime) / 1000 // 秒
      const speedMbps = (fileSize * 8) / duration // Mbps
      
      return {
        speed: Math.round(speedMbps * 10) / 10,
        bytes: fileSize * 1024 * 1024,
        duration
      }
    } catch (error) {
      console.error('上传速度测试失败:', error)
      throw new Error('上传速度测试失败')
    }
  },
  
  // 获取附近的测试服务器
  async getNearbyServers() {
    // 实际实现中会调用API获取附近的测试服务器
    return [
      { id: 1, name: '北京服务器', url: 'https://beijing.test.com', distance: 10 },
      { id: 2, name: '上海服务器', url: 'https://shanghai.test.com', distance: 20 },
      { id: 3, name: '广州服务器', url: 'https://guangzhou.test.com', distance: 30 }
    ]
  }
}

// 工具状态管理
const [toolState, setToolState] = useState({
  testServers: [],
  selectedServer: null,
  testStatus: 'idle', // 'idle', 'testing', 'completed', 'error'
  currentTest: 'ping', // 'ping', 'download', 'upload'
  testProgress: 0,
  testResults: {
    ping: null,
    download: null,
    upload: null,
    jitter: null,
    timestamp: null
  },
  testHistory: [],
  isTesting: false
})

// 初始化测试服务器
const initializeTestServers = async () => {
  try {
    const servers = await speedTest.getNearbyServers()
    setToolState(prev => ({
      ...prev,
      testServers: servers,
      selectedServer: servers[0]
    }))
  } catch (error) {
    console.error('获取测试服务器失败:', error)
    // 显示错误提示
  }
}

// 开始网速测试
const startSpeedTest = async () => {
  if (!toolState.selectedServer) return
  
  setToolState(prev => ({
    ...prev,
    testStatus: 'testing',
    testProgress: 0,
    testResults: { ping: null, download: null, upload: null, jitter: null, timestamp: null },
    isTesting: true
  }))
  
  try {
    // 测试Ping
    setToolState(prev => ({ ...prev, currentTest: 'ping' }))
    const pingResult = await speedTest.measurePing(toolState.selectedServer.url)
    setToolState(prev => ({
      ...prev,
      testProgress: 33,
      testResults: { ...prev.testResults, ping: pingResult }
    }))
    
    // 测试下载速度
    setToolState(prev => ({ ...prev, currentTest: 'download' }))
    const downloadResult = await speedTest.measureDownloadSpeed(toolState.selectedServer.url)
    setToolState(prev => ({
      ...prev,
      testProgress: 66,
      testResults: { ...prev.testResults, download: downloadResult }
    }))
    
    // 测试上传速度
    setToolState(prev => ({ ...prev, currentTest: 'upload' }))
    const uploadResult = await speedTest.measureUploadSpeed(toolState.selectedServer.url)
    setToolState(prev => ({
      ...prev,
      testProgress: 100,
      testResults: { ...prev.testResults, upload: uploadResult, timestamp: new Date().toISOString() },
      testStatus: 'completed'
    }))
    
    // 保存测试结果到历史记录
    saveTestResult({ ...toolState.testResults, ping: pingResult, download: downloadResult, upload: uploadResult, timestamp: new Date().toISOString() })
  } catch (error) {
    console.error('网速测试失败:', error)
    setToolState(prev => ({
      ...prev,
      testStatus: 'error',
      isTesting: false
    }))
    // 显示错误提示
  } finally {
    setToolState(prev => ({ ...prev, isTesting: false }))
  }
}

// 保存测试结果到历史记录
const saveTestResult = (result) => {
  setToolState(prev => {
    const updatedHistory = [result, ...prev.testHistory].slice(0, 10) // 保存最近10次测试结果
    // 保存到本地存储
    localStorage.setItem('speedTestHistory', JSON.stringify(updatedHistory))
    return { ...prev, testHistory: updatedHistory }
  })
}

// 加载测试历史记录
const loadTestHistory = () => {
  const history = localStorage.getItem('speedTestHistory')
  if (history) {
    setToolState(prev => ({ ...prev, testHistory: JSON.parse(history) }))
  }
}

// 组件挂载时初始化测试服务器和加载历史记录
useEffect(() => {
  initializeTestServers()
  loadTestHistory()
}, [])
```

## 4. 数据结构

```typescript
interface TestServer {
  id: number
  name: string
  url: string
  distance: number
}

interface SpeedTestResult {
  speed: number
  bytes: number
  duration: number
}

interface TestResults {
  ping: number | null
  download: SpeedTestResult | null
  upload: SpeedTestResult | null
  jitter: number | null
  timestamp: string | null
}

interface SpeedTestState {
  testServers: TestServer[]
  selectedServer: TestServer | null
  testStatus: 'idle' | 'testing' | 'completed' | 'error'
  currentTest: 'ping' | 'download' | 'upload'
  testProgress: number
  testResults: TestResults
  testHistory: TestResults[]
  isTesting: boolean
}
```

## 8. 总结

网速测试工具采用组件化设计，核心功能使用JavaScript的Fetch API和Performance API实现网络速度测量，无需额外的第三方库。工具提供Ping、下载速度和上传速度测试功能，支持测试服务器选择、测试进度显示和历史记录保存，帮助用户了解当前网络连接状态，适用于个人和专业场景。
