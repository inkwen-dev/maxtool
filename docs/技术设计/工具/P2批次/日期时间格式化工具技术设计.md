## 三、技术实现

### 1. 日期时间格式化实现

使用自定义的日期时间格式化函数：

```typescript
/**
 * 格式化日期时间
 * @param date - 日期时间对象
 * @param format - 格式模板
 * @param timezone - 时区
 * @returns 格式化后的日期时间字符串
 */
export function formatDateTime(date: Date, format: string, timezone: string = 'UTC'): string {
  // 实现日期时间格式化逻辑
  // 支持的格式占位符：YYYY, MM, DD, HH, hh, mm, ss, SSS, A, Z, ZZ
  
  const utcDate = new Date(date.toLocaleString('en-US', { timeZone: timezone }));
  
  const year = utcDate.getUTCFullYear();
  const month = utcDate.getUTCMonth() + 1;
  const day = utcDate.getUTCDate();
  const hours24 = utcDate.getUTCHours();
  const hours12 = hours24 % 12 || 12;
  const minutes = utcDate.getUTCMinutes();
  const seconds = utcDate.getUTCSeconds();
  const milliseconds = utcDate.getUTCMilliseconds();
  const ampm = hours24 < 12 ? 'AM' : 'PM';
  
  // 计算时区偏移
  const offsetMinutes = utcDate.getTimezoneOffset();
  const offsetHours = Math.floor(Math.abs(offsetMinutes) / 60);
  const offsetMinutesRemainder = Math.abs(offsetMinutes) % 60;
  const offsetSign = offsetMinutes <= 0 ? '+' : '-';
  const offset = `${offsetSign}${offsetHours.toString().padStart(2, '0')}:${offsetMinutesRemainder.toString().padStart(2, '0')}`;
  const offsetShort = `${offsetSign}${offsetHours.toString().padStart(2, '0')}${offsetMinutesRemainder.toString().padStart(2, '0')}`;
  
  // 替换格式占位符
  return format
    .replace('YYYY', year.toString())
    .replace('MM', month.toString().padStart(2, '0'))
    .replace('DD', day.toString().padStart(2, '0'))
    .replace('HH', hours24.toString().padStart(2, '0'))
    .replace('hh', hours12.toString().padStart(2, '0'))
    .replace('mm', minutes.toString().padStart(2, '0'))
    .replace('ss', seconds.toString().padStart(2, '0'))
    .replace('SSS', milliseconds.toString().padStart(3, '0'))
    .replace('A', ampm)
    .replace('ZZ', offset)
    .replace('Z', offsetShort);
}
```

### 2. 日期时间解析实现

使用自定义的日期时间解析函数：

```typescript
/**
 * 解析日期时间字符串
 * @param dateString - 日期时间字符串
 * @param timezone - 时区
 * @returns 日期时间对象
 */
export function parseDateTime(dateString: string, timezone: string = 'UTC'): Date {
  // 实现日期时间解析逻辑
  // 支持多种日期时间格式
  
  if (!dateString || typeof dateString !== 'string') {
    throw new Error('Invalid date string');
  }
  
  // 尝试使用Date构造函数解析
  const date = new Date(dateString);
  
  if (isNaN(date.getTime())) {
    // 尝试使用不同的解析策略
    // 例如：YYYY-MM-DD HH:mm:ss
    const isoRegex = /^(\d{4})-(\d{2})-(\d{2})\s+(\d{2}):(\d{2}):(\d{2})$/;
    const match = dateString.match(isoRegex);
    
    if (match) {
      const [, year, month, day, hours, minutes, seconds] = match;
      return new Date(`${year}-${month}-${day}T${hours}:${minutes}:${seconds}Z`);
    }
    
    throw new Error('Failed to parse date string');
  }
  
  return date;
}
```

/**
 * 计算相对时间
 * @param date - 日期时间对象
 * @returns 相对时间字符串
 */
export function calculateRelativeTime(date: Date): string {
  const now = new Date();
  const diffInMs = date.getTime() - now.getTime();
  const diffInSeconds = Math.round(diffInMs / 1000);
  const diffInMinutes = Math.round(diffInSeconds / 60);
  const diffInHours = Math.round(diffInMinutes / 60);
  const diffInDays = Math.round(diffInHours / 24);
  const diffInMonths = Math.round(diffInDays / 30);
  const diffInYears = Math.round(diffInDays / 365);
  
  const isPast = diffInMs < 0;
  const absSeconds = Math.abs(diffInSeconds);
  const absMinutes = Math.abs(diffInMinutes);
  const absHours = Math.abs(diffInHours);
  const absDays = Math.abs(diffInDays);
  const absMonths = Math.abs(diffInMonths);
  const absYears = Math.abs(diffInYears);
  
  const suffix = isPast ? '前' : '后';
  
  if (absYears > 0) {
    return `${absYears}年${suffix}`;
  } else if (absMonths > 0) {
    return `${absMonths}个月${suffix}`;
  } else if (absDays > 0) {
    return `${absDays}天${suffix}`;
  } else if (absHours > 0) {
    return `${absHours}小时${suffix}`;
  } else if (absMinutes > 0) {
    return `${absMinutes}分钟${suffix}`;
  } else if (absSeconds > 0) {
    return `${absSeconds}秒${suffix}`;
  } else {
    return '现在';
  }
}

// 批量格式化
const batchFormat = () => {
  if (!batchInput.trim()) {
    setBatchResults([])
    setError(null)
    return
  }

  try {
    setError(null)
    const dateStrings = batchInput.trim().split(/\n/)
    const results: string[] = []

    dateStrings.forEach(str => {
      if (str.trim()) {
        try {
          const date = parseDateTime(str.trim(), inputTimezone)
          if (validateDateTime(date)) {
            const formatted = formatDateTime(date, selectedPresetFormat, inputTimezone)
            results.push(formatted)
          } else {
            results.push(`无效的日期时间: ${str}`)
          }
        } catch (err) {
          results.push(`解析失败: ${str}`)
        }
      }
    })

    setBatchResults(results)
  } catch (err) {
    setError(err instanceof Error ? err.message : '批量格式化失败')
    setBatchResults([])
  }
