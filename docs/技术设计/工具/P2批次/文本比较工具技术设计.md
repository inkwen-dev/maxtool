# 文本比较工具技术设计文档

## 一、概述

本文档详细描述MaxTool 0.2版本中文本比较工具（Text Comparison）工具的技术设计，包括功能设计、技术实现、组件结构和代码复用等方面。设计严格遵循项目规范，注重代码复用、组件化和性能优化。

## 二、功能设计

### 1. 核心功能

#### 文本差异比较
- 比较两段文本的差异
- 支持行级比较
- 支持字符级比较

#### 差异高亮显示
- 新增内容高亮显示
- 删除内容高亮显示
- 修改内容高亮显示

#### 文本操作
- 支持从文件上传文本
- 支持清空文本内容
- 支持复制比较结果

### 2. 用户界面

#### 文本输入区域
- 双栏布局：左右两栏分别输入要比较的文本
- 文本输入框：支持多行文本输入
- 操作按钮：
  - 清空按钮：清空当前输入框内容
  - 上传按钮：从文件上传文本

#### 比较结果区域
- 差异高亮显示：使用不同颜色标识新增、删除和修改的内容
- 导航控件：快速跳转到下一个/上一个差异
- 比较模式选择：行级比较/字符级比较

#### 使用说明
- 提供简洁的使用步骤说明

## 三、技术设计

### 1. 技术选型

#### 核心库
- 原生JavaScript：用于文本差异比较的核心逻辑

#### 浏览器API
- Clipboard API：用于复制比较结果
- File API：用于文件上传功能

### 2. 组件结构

#### 主组件

##### TextComparison
- **位置**：`components/tools/TextComparison.tsx`
- **功能**：文本比较工具的主组件
- **结构**：
  ```tsx
  'use client'

  import { useState, useRef } from 'react'
  import Button from '@/components/ui/Button'
  import Textarea from '@/components/ui/Textarea'
  import { trackUsage } from '@/lib/tracking'
  import UsageInstructions from '@/components/shared/UsageInstructions'
  import { copyToClipboard } from '@/lib/clipboard-utils'

  // 定义比较模式
  type ComparisonMode = 'line' | 'character'

  // 定义差异类型
  type DiffType = 'equal' | 'insert' | 'delete'

  // 定义差异项
  interface DiffItem {
    type: DiffType
    content: string
  }

  // 定义差异结果
  interface DiffResult {
    left: DiffItem[]
    right: DiffItem[]
    hasDifferences: boolean
  }

  export default function TextComparison() {
    // State declarations
    const [text1, setText1] = useState<string>('')
    const [text2, setText2] = useState<string>('')
    const [mode, setMode] = useState<ComparisonMode>('line')
    const [diffResult, setDiffResult] = useState<DiffResult | null>(null)
    const [error, setError] = useState<string | null>(null)
    const [hasTracked, setHasTracked] = useState(false)
    const fileInputRef1 = useRef<HTMLInputElement>(null)
    const fileInputRef2 = useRef<HTMLInputElement>(null)

    // 初始化
    useState(() => {
      trackUsage('text-comparison')
      setHasTracked(true)
    }, [])

    // 比较文本
    const compareTexts = () => {
      try {
        setError(null)
        
        let result: DiffResult
        
        if (mode === 'line') {
          result = compareLines(text1, text2)
        } else {
          result = compareCharacters(text1, text2)
        }
        
        setDiffResult(result)
      } catch (err) {
        setError(err instanceof Error ? err.message : '比较失败')
        setDiffResult(null)
      }
    }

    // 行级比较
    const compareLines = (text1: string, text2: string): DiffResult => {
      const lines1 = text1.split('\n')
      const lines2 = text2.split('\n')
      
      const left: DiffItem[] = []
      const right: DiffItem[] = []
      let hasDifferences = false
      
      let i = 0, j = 0
      
      while (i < lines1.length || j < lines2.length) {
        if (i >= lines1.length) {
          // 文本2有额外的行
          right.push({ type: 'insert', content: lines2[j] })
          left.push({ type: 'equal', content: '' })
          hasDifferences = true
          j++
        } else if (j >= lines2.length) {
          // 文本1有额外的行
          left.push({ type: 'delete', content: lines1[i] })
          right.push({ type: 'equal', content: '' })
          hasDifferences = true
          i++
        } else if (lines1[i] === lines2[j]) {
          // 行内容相同
          left.push({ type: 'equal', content: lines1[i] })
          right.push({ type: 'equal', content: lines2[j] })
          i++
          j++
        } else {
          // 行内容不同
          left.push({ type: 'delete', content: lines1[i] })
          right.push({ type: 'insert', content: lines2[j] })
          hasDifferences = true
          i++
          j++
        }
      }
      
      return { left, right, hasDifferences }
    }

    // 字符级比较
    const compareCharacters = (text1: string, text2: string): DiffResult => {
      // 简化的字符比较算法
      const left: DiffItem[] = []
      const right: DiffItem[] = []
      let hasDifferences = false
      
      let i = 0, j = 0
      
      while (i < text1.length || j < text2.length) {
        if (i >= text1.length) {
          // 文本2有额外的字符
          right.push({ type: 'insert', content: text2[j] })
          left.push({ type: 'equal', content: '' })
          hasDifferences = true
          j++
        } else if (j >= text2.length) {
          // 文本1有额外的字符
          left.push({ type: 'delete', content: text1[i] })
          right.push({ type: 'equal', content: '' })
          hasDifferences = true
          i++
        } else if (text1[i] === text2[j]) {
          // 字符相同
          left.push({ type: 'equal', content: text1[i] })
          right.push({ type: 'equal', content: text2[j] })
          i++
          j++
        } else {
          // 字符不同
          left.push({ type: 'delete', content: text1[i] })
          right.push({ type: 'insert', content: text2[j] })
          hasDifferences = true
          i++
          j++
        }
      }
      
      return { left, right, hasDifferences }
    }

    // 清空文本
    const clearText = (index: number) => {
      if (index === 1) {
        setText1('')
      } else {
        setText2('')
      }
      setDiffResult(null)
    }

    // 上传文件
    const handleFileUpload = (index: number) => {
      const fileInput = index === 1 ? fileInputRef1.current : fileInputRef2.current
      if (fileInput) {
        fileInput.click()
      }
    }

    // 处理文件选择
    const handleFileChange = (e: React.ChangeEvent<HTMLInputElement>, index: number) => {
      const file = e.target.files?.[0]
      if (file) {
        const reader = new FileReader()
        reader.onload = (event) => {
          const content = event.target?.result as string
          if (index === 1) {
            setText1(content)
          } else {
            setText2(content)
          }
          setDiffResult(null)
        }
        reader.onerror = () => {
          setError('文件读取失败')
        }
        reader.readAsText(file)
      }
      // 重置文件输入，以便可以重新选择相同的文件
      e.target.value = ''
    }

    // 复制结果
    const copyResults = () => {
      if (diffResult) {
        const leftContent = diffResult.left.map(item => item.content).join('')
        const rightContent = diffResult.right.map(item => item.content).join('')
        const result = `左侧文本:\n${leftContent}\n\n右侧文本:\n${rightContent}`
        copyToClipboard(result)
        // 显示复制成功提示
      }
    }

    // 获取差异类型的CSS类名
    const getDiffClass = (type: DiffType): string => {
      switch (type) {
        case 'insert':
          return 'bg-green-100 border-l-4 border-green-500'
        case 'delete':
          return 'bg-red-100 border-l-4 border-red-500'
        case 'equal':
          return ''
      }
    }

    // Render
    return (
      <div className="space-y-6">
        {/* Header */}
        <div className="space-y-4">
          <h2 className="text-xl font-semibold">文本比较工具</h2>
        </div>

        {/* Comparison Mode */}
        <div className="space-y-4">
          <div className="flex border-b">
            <button
              className={`px-5 py-2 border-b-2 font-medium ${mode === 'line' ? 'border-primary text-primary' : 'border-transparent text-text-secondary hover:text-text-primary'}`}
              onClick={() => setMode('line')}
            >
              行级比较
            </button>
            <button
              className={`px-5 py-2 border-b-2 font-medium ${mode === 'character' ? 'border-primary text-primary' : 'border-transparent text-text-secondary hover:text-text-primary'}`}
              onClick={() => setMode('character')}
            >
              字符级比较
            </button>
          </div>
        </div>

        {/* Text Inputs */}
        <div className="space-y-6">
          <div className="grid grid-cols-2 gap-6">
            {/* Text 1 */}
            <div className="space-y-3">
              <div className="flex justify-between items-center">
                <h3 className="text-lg font-semibold">文本1</h3>
                <div className="flex space-x-2">
                  <Button onClick={() => handleFileUpload(1)} variant="secondary" size="sm">
                    上传
                  </Button>
                  <Button onClick={() => clearText(1)} disabled={!text1} variant="secondary" size="sm">
                    清空
                  </Button>
                </div>
              </div>
              <Textarea
                value={text1}
                onChange={(e) => setText1(e.target.value)}
                placeholder="请输入要比较的文本1..."
                className="min-h-[300px]"
              />
              <input
                type="file"
                ref={fileInputRef1}
                onChange={(e) => handleFileChange(e, 1)}
                accept=".txt,.md,.html,.css,.js,.json,.xml"
                className="hidden"
              />
            </div>
            
            {/* Text 2 */}
            <div className="space-y-3">
              <div className="flex justify-between items-center">
                <h3 className="text-lg font-semibold">文本2</h3>
                <div className="flex space-x-2">
                  <Button onClick={() => handleFileUpload(2)} variant="secondary" size="sm">
                    上传
                  </Button>
                  <Button onClick={() => clearText(2)} disabled={!text2} variant="secondary" size="sm">
                    清空
                  </Button>
                </div>
              </div>
              <Textarea
                value={text2}
                onChange={(e) => setText2(e.target.value)}
                placeholder="请输入要比较的文本2..."
                className="min-h-[300px]"
              />
              <input
                type="file"
                ref={fileInputRef2}
                onChange={(e) => handleFileChange(e, 2)}
                accept=".txt,.md,.html,.css,.js,.json,.xml"
                className="hidden"
              />
            </div>
          </div>
          
          {/* Compare Button */}
          <div className="flex justify-center">
            <Button onClick={compareTexts} disabled={!text1 || !text2} variant="primary" size="lg">
              比较文本
            </Button>
          </div>
        </div>

        {/* Error Message */}
        {error && (
          <div className="p-2 bg-red-50 border border-red-200 rounded text-red-700">
            {error}
          </div>
        )}

        {/* Comparison Result */}
        {diffResult && (
          <div className="space-y-4">
            <div className="flex justify-between items-center">
              <h3 className="text-lg font-semibold">比较结果</h3>
              <Button onClick={copyResults} variant="secondary" size="sm">
                复制结果
              </Button>
            </div>
            
            <div className="p-2 bg-yellow-50 border border-yellow-200 rounded text-yellow-700">
              {diffResult.hasDifferences ? '发现差异' : '两段文本完全相同'}
            </div>
            
            <div className="grid grid-cols-2 gap-6">
              {/* Left Result */}
              <div className="space-y-3">
                <h4 className="text-md font-semibold">文本1</h4>
                <div className="p-4 border border-border-secondary bg-bg-secondary rounded-lg max-h-[500px] overflow-auto">
                  {mode === 'line' ? (
                    <pre className="whitespace-pre-wrap">
                      {diffResult.left.map((item, index) => (
                        <div key={index} className={`py-1 pl-2 ${getDiffClass(item.type)}`}>
                          {item.content || <span className="text-text-tertiary italic">（空行）</span>}
                        </div>
                      ))}
                    </pre>
                  ) : (
                    <div className="flex flex-col">
                      {diffResult.left.map((item, index) => (
                        <span key={index} className={`inline-block ${getDiffClass(item.type)} px-1 py-0.5 rounded`}>
                          {item.content}
                        </span>
                      ))}
                    </div>
                  )}
                </div>
              </div>
              
              {/* Right Result */}
              <div className="space-y-3">
                <h4 className="text-md font-semibold">文本2</h4>
                <div className="p-4 border border-border-secondary bg-bg-secondary rounded-lg max-h-[500px] overflow-auto">
                  {mode === 'line' ? (
                    <pre className="whitespace-pre-wrap">
                      {diffResult.right.map((item, index) => (
                        <div key={index} className={`py-1 pl-2 ${getDiffClass(item.type)}`}>
                          {item.content || <span className="text-text-tertiary italic">（空行）</span>}
                        </div>
                      ))}
                    </pre>
                  ) : (
                    <div className="flex flex-col">
                      {diffResult.right.map((item, index) => (
                        <span key={index} className={`inline-block ${getDiffClass(item.type)} px-1 py-0.5 rounded`}>
                          {item.content}
                        </span>
                      ))}
                    </div>
                  )}
                </div>
              </div>
            </div>
          </div>
        )}

        {/* Usage Instructions */}
        <UsageInstructions
          items={[
            '在文本1和文本2输入框中输入要比较的内容',
            '可选择上传文件或直接输入文本',
            '选择比较模式：行级比较或字符级比较',
            '点击"比较文本"按钮查看差异',
            '差异将以不同颜色高亮显示：红色表示删除，绿色表示新增',
            '点击"复制结果"按钮复制比较结果'
          ]}
        />
      </div>
    )
  }
  ```
