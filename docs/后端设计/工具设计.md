### 1. 抽签工具（Draw lots）⭐⭐

**路径：** `/tools/draw-lots`  
**工具名称：** `draw-lots`

**功能：**
- 导入名单（手动输入，每行一个）
- 随机抽取
- 已中奖者不参与下一轮
- 多轮抽奖支持
- 显示已中奖列表

**实现方案：**

```typescript
// app/tools/draw-lots/page.tsx
'use client'

import { useState } from 'react'
import { trackUsage } from '@/lib/tracking'
import { saveToStorage, getFromStorage } from '@/lib/storage'

export default function DrawLots() {
  const [names, setNames] = useState<string[]>([])
  const [winners, setWinners] = useState<string[]>([])
  const [currentWinner, setCurrentWinner] = useState<string | null>(null)

  // 从localStorage加载名单
  useState(() => {
    const saved = getFromStorage<string[]>('draw-lots-names')
    if (saved) setNames(saved)
  }, [])

  // 导入名单
  const handleImport = (text: string) => {
    const list = text.split('\n').filter(n => n.trim())
    setNames(list)
    saveToStorage('draw-lots-names', list)
  }

  // 抽取
  const draw = () => {
    const available = names.filter(n => !winners.includes(n))
    if (available.length === 0) {
      alert('所有人已中奖！')
      return
    }
    
    const randomIndex = Math.floor(Math.random() * available.length)
    const winner = available[randomIndex]
    
    setCurrentWinner(winner)
    setWinners([...winners, winner])
    trackUsage('draw-lots')
  }

  // 重置
  const reset = () => {
    setWinners([])
    setCurrentWinner(null)
  }

  return (
    <div className="container mx-auto p-8">
      <h1 className="text-3xl font-bold mb-4">抽签工具</h1>
      
      <div className="mb-4">
        <textarea 
          className="w-full p-2 border rounded"
          placeholder="输入名单，每行一个"
          rows={10}
          onChange={(e) => handleImport(e.target.value)}
          defaultValue={names.join('\n')}
        />
      </div>

      <div className="flex gap-4 mb-4">
        <button 
          className="px-4 py-2 bg-blue-500 text-white rounded"
          onClick={draw}
          disabled={names.length === 0}
        >
          抽取
        </button>
        <button 
          className="px-4 py-2 bg-gray-500 text-white rounded"
          onClick={reset}
        >
          重置
        </button>
      </div>

      {currentWinner && (
        <div className="mb-4 p-4 bg-yellow-100 rounded text-center">
          <div className="text-2xl font-bold">{currentWinner}</div>
        </div>
      )}

      {winners.length > 0 && (
        <div>
          <h3 className="text-xl font-bold mb-2">已中奖：</h3>
          <ul className="list-disc list-inside">
            {winners.map((w, i) => (
              <li key={i}>{w}</li>
            ))}
          </ul>
        </div>
      )}
    </div>
  )
}
```

**技术要点：**
- 使用 `Math.random()` 随机选择
- 使用 `filter` 排除已中奖者
- 使用 `localStorage` 保存名单
- 输入验证（确保有名单）

### 2. 随机数生成器（Random number generator）⭐⭐

**路径：** `/tools/random-number`  
**工具名称：** `random-number`

**功能：**
- 设置最小值和最大值
- 生成随机数
- 显示结果
- 输入验证

**实现方案：**

```typescript
// app/tools/random-number/page.tsx
'use client'

import { useState } from 'react'
import { trackUsage } from '@/lib/tracking'

export default function RandomNumber() {
  const [min, setMin] = useState(1)
  const [max, setMax] = useState(100)
  const [result, setResult] = useState<number | null>(null)

  const generate = () => {
    if (min >= max) {
      alert('最小值必须小于最大值')
      return
    }
    
    const random = Math.floor(Math.random() * (max - min + 1)) + min
    setResult(random)
    trackUsage('random-number')
  }

  return (
    <div className="container mx-auto p-8">
      <h1 className="text-3xl font-bold mb-4">随机数生成器</h1>
      
      <div className="flex items-center gap-4 mb-4">
        <input 
          type="number" 
          className="px-4 py-2 border rounded"
          value={min} 
          onChange={(e) => setMin(Number(e.target.value))} 
        />
        <span>到</span>
        <input 
          type="number" 
          className="px-4 py-2 border rounded"
          value={max} 
          onChange={(e) => setMax(Number(e.target.value))} 
        />
      </div>

      <button 
        className="px-4 py-2 bg-blue-500 text-white rounded"
        onClick={generate}
      >
        生成
      </button>

      {result !== null && (
        <div className="mt-4 p-4 bg-blue-100 rounded text-center">
          <div className="text-3xl font-bold">{result}</div>
        </div>
      )}
    </div>
  )
}
```

**技术要点：**
- `Math.floor(Math.random() * (max - min + 1)) + min` 公式
- 输入验证（确保 min < max）

### 3. 抛硬币（Flip a coin）⭐⭐

**路径：** `/tools/flip-coin`  
**工具名称：** `flip-coin`

**功能：**
- 点击抛硬币
- 3D动画效果
- 显示结果（正面/反面）

**实现方案：**

```typescript
// app/tools/flip-coin/page.tsx
'use client'

import { useState } from 'react'
import { trackUsage } from '@/lib/tracking'

export default function FlipCoin() {
  const [result, setResult] = useState<'heads' | 'tails' | null>(null)
  const [flipping, setFlipping] = useState(false)

  const flip = () => {
    setFlipping(true)
    setResult(null)
    
    setTimeout(() => {
      const random = Math.random()
      setResult(random < 0.5 ? 'heads' : 'tails')
      setFlipping(false)
      trackUsage('flip-coin')
    }, 1000)
  }

  return (
    <div className="container mx-auto p-8 text-center">
      <h1 className="text-3xl font-bold mb-4">抛硬币</h1>
      
      <div className={`coin ${flipping ? 'flipping' : ''} mb-4`}>
        {result === 'heads' && <div className="text-6xl">🪙 正面</div>}
        {result === 'tails' && <div className="text-6xl">🪙 反面</div>}
        {!result && <div className="text-6xl">🪙</div>}
      </div>

      <button 
        className="px-4 py-2 bg-blue-500 text-white rounded"
        onClick={flip}
        disabled={flipping}
      >
        抛硬币
      </button>
    </div>
  )
}
```

### 6. 随机团队生成器（Random Team Generator）⭐⭐

**路径：** `/tools/random-team`  
**工具名称：** `random-team`

**功能：**
- 输入人员名单
- 设置团队数量
- 随机分配团队

**实现方案：**

```typescript
// app/tools/random-team/page.tsx
'use client'

import { useState } from 'react'
import { trackUsage } from '@/lib/tracking'

export default function RandomTeam() {
  const [names, setNames] = useState<string[]>([])
  const [teamCount, setTeamCount] = useState(2)
  const [teams, setTeams] = useState<string[][]>([])

  const handleInput = (text: string) => {
    const list = text.split('\n').filter(n => n.trim())
    setNames(list)
  }

  const generate = () => {
    if (names.length === 0) {
      alert('请先输入人员名单')
      return
    }
    
    const shuffled = [...names].sort(() => Math.random() - 0.5)
    const result: string[][] = Array(teamCount).fill(null).map(() => [])
    
    shuffled.forEach((name, index) => {
      result[index % teamCount].push(name)
    })
    
    setTeams(result)
    trackUsage('random-team')
  }

  return (
    <div className="container mx-auto p-8">
      <h1 className="text-3xl font-bold mb-4">随机团队生成器</h1>
      
      <textarea 
        className="w-full p-2 border rounded mb-4"
        placeholder="输入人员名单，每行一个"
        rows={10}
        onChange={(e) => handleInput(e.target.value)}
      />

      <div className="mb-4">
        <label>团队数量：</label>
        <input 
          type="number" 
          className="px-4 py-2 border rounded"
          value={teamCount} 
          min={2}
          onChange={(e) => setTeamCount(Number(e.target.value))} 
        />
      </div>

      <button 
        className="px-4 py-2 bg-blue-500 text-white rounded mb-4"
        onClick={generate}
      >
        生成团队
      </button>

      {teams.map((team, i) => (
        <div key={i} className="mb-4 p-4 border rounded">
          <h3 className="text-xl font-bold mb-2">团队{i + 1}</h3>
          <ul className="list-disc list-inside">
            {team.map(name => <li key={name}>{name}</li>)}
          </ul>
        </div>
      ))}
    </div>
  )
}
```

**技术要点：**
- 数组随机排序：`sort(() => Math.random() - 0.5)`
- 轮询分配算法

```typescript
import { useState, useEffect } from 'react'
```

### 9. 倒计时器（Countdown timer）⭐⭐

**路径：** `/tools/countdown-timer`  
**工具名称：** `countdown-timer`

**功能：**
- 设置倒计时时间
- 开始/暂停/重置
- 显示剩余时间
- 时间到提醒

**实现方案：**

```typescript
// app/tools/countdown-timer/page.tsx
'use client'

import { useState, useEffect, useRef } from 'react'
import { trackUsage } from '@/lib/tracking'
import { formatTime } from '@/lib/utils'

export default function CountdownTimer() {
  const [minutes, setMinutes] = useState(5)
  const [seconds, setSeconds] = useState(0)
  const [isRunning, setIsRunning] = useState(false)
  const intervalRef = useRef<NodeJS.Timeout | null>(null)

  useEffect(() => {
    if (isRunning) {
      intervalRef.current = setInterval(() => {
        setSeconds(prev => {
          if (prev === 0) {
            if (minutes === 0) {
              setIsRunning(false)
              alert('时间到！')
              return 0
            }
            setMinutes(prev => prev - 1)
            return 59
          }
          return prev - 1
        })
      }, 1000)
    } else {
      if (intervalRef.current) {
        clearInterval(intervalRef.current)
      }
    }

    return () => {
      if (intervalRef.current) {
        clearInterval(intervalRef.current)
      }
    }
  }, [isRunning, minutes])

  const start = () => {
    setIsRunning(true)
    trackUsage('countdown-timer')
  }

  const pause = () => setIsRunning(false)

  const reset = () => {
    setIsRunning(false)
    setMinutes(5)
    setSeconds(0)
  }

  const totalSeconds = minutes * 60 + seconds

  return (
    <div className="container mx-auto p-8 text-center">
      <h1 className="text-3xl font-bold mb-4">倒计时器</h1>
      
      <div className="mb-4">
        <input 
          type="number" 
          className="px-4 py-2 border rounded"
          value={minutes} 
          min={0}
          onChange={(e) => setMinutes(Number(e.target.value))} 
        />
        <span className="mx-2">:</span>
        <input 
          type="number" 
          className="px-4 py-2 border rounded"
          value={seconds} 
          min={0}
          max={59}
          onChange={(e) => setSeconds(Number(e.target.value))} 
        />
      </div>

      <div className="text-6xl font-mono mb-4">
        {formatTime(totalSeconds)}
      </div>

      <div className="flex gap-4 justify-center">
        <button 
          className="px-4 py-2 bg-blue-500 text-white rounded"
          onClick={start}
          disabled={isRunning}
        >
          开始
        </button>
        <button 
          className="px-4 py-2 bg-yellow-500 text-white rounded"
          onClick={pause}
          disabled={!isRunning}
        >
          暂停
        </button>
        <button 
          className="px-4 py-2 bg-gray-500 text-white rounded"
          onClick={reset}
        >
          重置
        </button>
      </div>
    </div>
  )
}
```

**技术要点：**
- `useRef` 保存定时器引用
- `useEffect` 清理定时器
- 时间计算逻辑

### 11. 番茄工作法计时器（Pomodoro Timer）⭐⭐⭐

**路径：** `/tools/pomodoro`  
**工具名称：** `pomodoro`

**功能：**
- 25分钟工作 + 5分钟休息
- 开始/暂停/重置
- 工作/休息状态切换
- 提醒音效（可选）

**实现方案：**

```typescript
// app/tools/pomodoro/page.tsx
'use client'

import { useState, useEffect, useRef } from 'react'
import { trackUsage } from '@/lib/tracking'
import { formatTime } from '@/lib/utils'

export default function Pomodoro() {
  const [mode, setMode] = useState<'work' | 'break'>('work')
  const [time, setTime] = useState(25 * 60)
  const [isRunning, setIsRunning] = useState(false)
  const intervalRef = useRef<NodeJS.Timeout | null>(null)

  useEffect(() => {
    if (isRunning) {
      intervalRef.current = setInterval(() => {
        setTime(prev => {
          if (prev === 0) {
            if (mode === 'work') {
              setMode('break')
              return 5 * 60
            } else {
              setMode('work')
              return 25 * 60
            }
          }
          return prev - 1
        })
      }, 1000)
    } else {
      if (intervalRef.current) {
        clearInterval(intervalRef.current)
      }
    }

    return () => {
      if (intervalRef.current) {
        clearInterval(intervalRef.current)
      }
    }
  }, [isRunning, mode])

  return (
    <div className="container mx-auto p-8 text-center">
      <h1 className="text-3xl font-bold mb-4">番茄工作法</h1>
      
      <div className={`mb-4 text-2xl ${mode === 'work' ? 'text-red-500' : 'text-green-500'}`}>
        {mode === 'work' ? '工作' : '休息'}
      </div>

      <div className="text-6xl font-mono mb-4">
        {formatTime(time)}
      </div>

      <button 
        className="px-4 py-2 bg-blue-500 text-white rounded"
        onClick={() => {
          setIsRunning(!isRunning)
          if (!isRunning) trackUsage('pomodoro')
        }}
      >
        {isRunning ? '暂停' : '开始'}
      </button>
    </div>
  )
}
```

**技术要点：**
- 状态机（工作/休息）
- 自动切换模式

```typescript
'use client'

import { useState } from 'react'
import { trackUsage } from '@/lib/tracking'
import { copyToClipboard } from '@/lib/utils'

const fontStyles: Record<string, (text: string) => string> = {
  outline: (text: string) => {
    // Unicode字符映射示例
    return text.split('').map(c => {
      const code = c.charCodeAt(0)
      if (code >= 65 && code <= 90) {
        return String.fromCharCode(0x1D400 + code - 65)
      }
      return c
    }).join('')
  },
  bubble: (text: string) => {
    // 气泡字体示例
    return text.split('').map(c => {
      const code = c.charCodeAt(0)
      if (code >= 65 && code <= 90) {
        return String.fromCharCode(0x1F130 + code - 65)
      }
      return c
    }).join('')
  }
}

export default function FontGenerator() {
  const [input, setInput] = useState('')
  const [style, setStyle] = useState('outline')
  const [output, setOutput] = useState('')

  const generate = () => {
    const result = fontStyles[style]?.(input) || input
    setOutput(result)
    trackUsage('font-generator')
  }

  const copy = async () => {
    await copyToClipboard(output)
    alert('已复制')
  }

  return (
    <div className="container mx-auto p-8">
      <h1 className="text-3xl font-bold mb-4">字体生成器</h1>
      
      <textarea 
        className="w-full p-2 border rounded mb-4"
        value={input}
        onChange={(e) => setInput(e.target.value)}
        rows={5}
      />

      <select 
        className="px-4 py-2 border rounded mb-4"
        value={style}
        onChange={(e) => setStyle(e.target.value)}
      >
        <option value="outline">轮廓字体</option>
        <option value="bubble">气泡字体</option>
      </select>

      <button 
        className="px-4 py-2 bg-blue-500 text-white rounded mb-4"
        onClick={generate}
      >
        生成
      </button>

      {output && (
        <div>
          <div className="p-4 border rounded mb-4">{output}</div>
          <button 
            className="px-4 py-2 bg-green-500 text-white rounded"
            onClick={copy}
          >
            复制
          </button>
        </div>
      )}
    </div>
  )
}
```

```typescript
// app/tools/text-to-speech/page.tsx
'use client'

import { useState, useEffect } from 'react'
import { trackUsage } from '@/lib/tracking'

export default function TextToSpeech() {
  const [text, setText] = useState('')
  const [voice, setVoice] = useState<string>('')
  const [voices, setVoices] = useState<SpeechSynthesisVoice[]>([])

  useEffect(() => {

      setVoices(speechSynthesis.getVoices())
    }

    speechSynthesis.onvoiceschanged = loadVoices
  }, [])

    if (!text) return
    
    const utterance = new SpeechSynthesisUtterance(text)
    
    if (voice) {
      const selectedVoice = voices.find(v => v.name === voice)
      if (selectedVoice) {
        utterance.voice = selectedVoice
      }
    }
    
    speechSynthesis.speak(utterance)
    trackUsage('text-to-speech')
  }

  return (
    <div className="container mx-auto p-8">
      <h1 className="text-3xl font-bold mb-4">文本转语音</h1>

        className="w-full p-2 border rounded mb-4"
        value={text}
        onChange={(e) => setText(e.target.value)}
        rows={5}
      />

      <select 
        className="px-4 py-2 border rounded mb-4"
        value={voice}
        onChange={(e) => setVoice(e.target.value)}
      >
        <option value="">默认语音</option>
        {voices.map(v => (
          <option key={v.name} value={v.name}>{v.name}</option>
        ))}
      </select>

      <button 
        className="px-4 py-2 bg-blue-500 text-white rounded"
        onClick={speak}
      >
```
